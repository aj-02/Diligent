METHOD if_ex_hcmfab_common~get_onbehalf_employees.

* Determine the set of employees which can be accessed on behalf of the current user

  DATA: lt_onbehalf_pernrs  TYPE pernr_tab,
        lv_onbehalf_pernr   TYPE pernr_d,
        lv_onbehalf_enabled TYPE boole_d,
        lv_hide_pernr       TYPE boole_d,
        lv_hide_pic         TYPE boole_d.

  FIELD-SYMBOLS: <ls_entity> TYPE hcmfab_s_onbehalf_employee.


  CLEAR et_onbehalf_employees.

* check whether on-behalf is active at all for the given employee number
  me->get_configuration(
    EXPORTING
      iv_application_id       = iv_application_id
      iv_employee_number      = iv_employee_number
    IMPORTING
      ev_hide_employee_number = lv_hide_pernr
      ev_enable_onbehalf      = lv_onbehalf_enabled
  ).
  IF lv_onbehalf_enabled = abap_false.
    RETURN.
  ENDIF.

* Use the direct reports of the current user
  me->get_direct_reports(
    EXPORTING
      iv_application_id = iv_application_id
      iv_manager_pernr  = iv_employee_number
    IMPORTING
      et_direct_reports = lt_onbehalf_pernrs
  ).

* for the given PERNRs read the corresponding data
  LOOP AT lt_onbehalf_pernrs INTO lv_onbehalf_pernr.
    APPEND INITIAL LINE TO et_onbehalf_employees ASSIGNING <ls_entity>.

    <ls_entity>-application_id         = iv_application_id.
    <ls_entity>-employee_number        = lv_onbehalf_pernr.
    <ls_entity>-employee_name          = go_employee_api->get_name( iv_pernr         = lv_onbehalf_pernr
                                                                    iv_no_auth_check = abap_true ).
    <ls_entity>-employee_sortname      = go_employee_api->get_sortable_name( iv_pernr         = lv_onbehalf_pernr
                                                                             iv_no_auth_check = abap_true ).
    <ls_entity>-requester_number       = iv_employee_number.
    <ls_entity>-employee_category      = gc_employee_category-direct_report.

*   Leave default version id empty. It will then automatically be calculated later on, but only for the relevant employees.
    CLEAR <ls_entity>-default_version_id.

    IF lv_hide_pernr = abap_true.
      "if no employee number should be shown we display the employee´s position to better distinguish the employees
      <ls_entity>-description =
         go_employee_api->get_employee_details( iv_application_id   = iv_application_id
                                                iv_pernr            = lv_onbehalf_pernr )-employee_position_longtext.
    ENDIF.

*   get the configuration info for the "on-behalf employee"
    CLEAR lv_hide_pernr.
    me->get_configuration(
      EXPORTING
        iv_application_id            = iv_application_id
        iv_employee_number           = lv_onbehalf_pernr
      IMPORTING
        ev_hide_employee_picture     = lv_hide_pic
        ev_hide_employee_number      = lv_hide_pernr
        ev_show_empl_number_wo_zeros = <ls_entity>-show_employee_number_wo_zeros
    ).

    <ls_entity>-show_employee_number      = boolc( lv_hide_pernr = abap_false ).
    <ls_entity>-show_employee_picture     = boolc( lv_hide_pic = abap_false ).

    "in SAP-standard, managers are not allowed to change the employee picture of the onBehalf-employees
    <ls_entity>-is_change_picture_enabled = abap_false.

    "display the personal number in the dialog (additional properties can be added as comma separated list)
    <ls_entity>-displayed_properties = 'EmployeeId'.

  ENDLOOP.

  SORT et_onbehalf_employees BY employee_sortname.

ENDMETHOD.

                                             iv_telnumber = ls_telephone-tel_number
                                             iv_extension = ls_telephone-tel_extens ).
    IF lv_phone_formatted IS NOT INITIAL.
      es_employee_details-work_phone = lv_phone_formatted.
    ELSE.
      es_employee_details-work_phone = ls_telephone-tel_number && '-' && ls_telephone-tel_extens.
    ENDIF.
  ENDIF.

  "Office E-Mail address
  CALL METHOD cl_hcmfab_utilities=>read_infotype_record
    EXPORTING
      iv_pernr       = iv_pernr
      iv_infty       = '0105'
      iv_subty       = '0010'
    IMPORTING
      es_pnnnn       = ls_p0105
      ev_data_exists = lv_data_exists.

  IF lv_data_exists = abap_false.
    CALL METHOD cl_hcmfab_utilities=>read_infotype_record
      EXPORTING
        iv_pernr       = iv_pernr
        iv_infty       = '0105'
        iv_subty       = 'MAIL'
      IMPORTING
        es_pnnnn       = ls_p0105
        ev_data_exists = lv_data_exists.
  ENDIF.

  IF lv_data_exists = abap_true.
    IF ls_p0105-usrid_long IS NOT INITIAL.
      es_employee_details-work_email = ls_p0105-usrid_long.
    ELSE.
      es_employee_details-work_email = ls_p0105-usrid.
    ENDIF.
  ENDIF.

***********************************************************************
* local time info
***********************************************************************
  es_employee_details-local_timezone = go_employee_api->get_timezone_from_user( es_employee_details-user_id ).
  IF es_employee_details-local_timezone IS NOT INITIAL.
    "get timezone text
    CALL FUNCTION 'TZON_GET_TIMEZONE_TEXT'
      EXPORTING
        if_timezone       = es_employee_details-local_timezone
      IMPORTING
        ef_timezone_text  = es_employee_details-local_timezone_text
      EXCEPTIONS
        no_text_available = 1
        OTHERS            = 2.
    IF sy-subrc <> 0.
      CLEAR es_employee_details-local_timezone_text.
    ENDIF.

    "get UTC-timestamp and convert it into employees local timezone
    GET TIME STAMP FIELD lv_timestamp.
    CONVERT TIME STAMP lv_timestamp
                 TIME ZONE es_employee_details-local_timezone
                 INTO DATE lv_date TIME lv_time
                 DAYLIGHT SAVING TIME lv_dst.

    "get employees timezone offset to UTC
    CALL FUNCTION 'TZON_GET_OFFSET'
      EXPORTING
        if_timezone      = es_employee_details-local_timezone
        if_local_date    = lv_date
        if_local_time    = lv_time
      IMPORTING
        ef_utcdiff       = es_employee_details-local_timezone_offset
        ef_utcsign       = es_employee_details-local_timezone_offset_sign
        ef_is_in_dst     = es_employee_details-local_timezone_offset_is_dst
      EXCEPTIONS
        conversion_error = 1
        OTHERS           = 2.
    IF sy-subrc <> 0.
      CLEAR: es_employee_details-local_timezone_offset,
             es_employee_details-local_timezone_offset_sign,
             es_employee_details-local_timezone_offset_is_dst.
    ENDIF.
  ENDIF.

***********************************************************************
* direct reports count
***********************************************************************
  IF go_employee_api->is_manager( iv_application_id = iv_application_id
                                  iv_pernr          = iv_pernr ) = abap_true.
    es_employee_details-direct_reports_count =
      lines( go_employee_api->get_direct_reports( iv_application_id = iv_application_id
                                                  iv_manager_pernr  = iv_pernr ) ).
  ENDIF.

ENDMETHOD.

METHOD if_ex_hcmfab_common~get_office_address.
* in SAP-standard we retrieve the office address from the org.management (OM)
* infotype 1028 for the position (S) or the org.unit (O)
* --> Building and Room Number from IT1028 are overruled by IT0032 (if available)

* OPTION for customers:
* if needed, customers can also retrieve the office address from
* the user details (SU01) by using the following method:
* CL_HCMFAB_FB_COMMON->GET_OFFICE_ADDRESS_FROM_USER
* (customer-own implementation class can inherit from CL_HCMFAB_FB_COMMON
*  with re-definition of only the needed methods)

  DATA: ls_hrobject TYPE hrobject,
        ls_p0001    TYPE p0001,
        ls_p0032    TYPE p0032.


  CLEAR es_office_address.

* first try to get the office address from the position
  ls_hrobject-plvar = cl_hcmfab_employee_api=>gv_active_plvar.
  ls_hrobject-otype = gc_otype-position.
  ls_hrobject-objid = go_employee_api->get_position( iv_pernr ).

  es_office_address = me->get_office_address_from_org( ls_hrobject ).

  IF es_office_address IS INITIAL.
    CALL METHOD cl_hcmfab_utilities=>read_infotype_record
      EXPORTING
        iv_pernr         = iv_pernr
        iv_infty         = '0001'
        iv_no_auth_check = abap_true
      IMPORTING
        es_pnnnn         = ls_p0001.
    IF ls_p0001 IS INITIAL.
      RETURN.
    ENDIF.

*   then try to get the office address from the org.unit
    ls_hrobject-otype = gc_otype-orgunit.
    ls_hrobject-objid = ls_p0001-orgeh.

    es_office_address = me->get_office_address_from_org( ls_hrobject ).
  ENDIF.

  "Office Location (Building and Room Number)
  CALL METHOD cl_hcmfab_utilities=>read_infotype_record
    EXPORTING
      iv_pernr = iv_pernr
      iv_infty = '0032'
    IMPORTING
      es_pnnnn = ls_p0032.

  IF ls_p0032-zimnr IS NOT INITIAL.
    es_office_address-office_room = ls_p0032-zimnr.
  ENDIF.
  IF ls_p0032-gebnr IS NOT INITIAL.
    es_office_address-office_building = ls_p0032-gebnr.
  ENDIF.

ENDMETHOD.

METHOD buffer_full_org_address_data.

  DATA: lt_t777a TYPE SORTED TABLE OF t777a WITH UNIQUE KEY build.

  FIELD-SYMBOLS: <ls_hri1028> TYPE p1028,
                 <ls_t777a>   TYPE t777a.


* check if buffer was already filled
  IF gt_hrp1028 IS NOT INITIAL.
    RETURN.
  ENDIF.

* buffer org.address data from infotype 1028
  CALL FUNCTION 'RH_READ_INFTY'
    EXPORTING
      with_stru_auth       = abap_true
      plvar                = cl_hcmfab_employee_api=>gv_active_plvar
      infty                = '1028'
      begda                = sy-datlo                      "MELN2857008
      endda                = sy-datlo
      sort                 = abap_true
    TABLES
      innnn                = gt_hrp1028
    EXCEPTIONS
      all_infty_with_subty = 1
      nothing_found        = 2
      no_objects           = 3
      wrong_condition      = 4
      wrong_parameters     = 5
      OTHERS               = 6.
  IF sy-subrc <> 0.
    CLEAR gt_hrp1028.
    RETURN.
  ENDIF.

  DELETE gt_hrp1028 WHERE otype <> if_hcmfab_constants=>gc_otype-orgunit
                      AND otype <> if_hcmfab_constants=>gc_otype-position.

  SELECT * FROM t777a INTO TABLE lt_t777a.
  LOOP AT gt_hrp1028 ASSIGNING <ls_hri1028> WHERE build IS NOT INITIAL.

    READ TABLE lt_t777a WITH TABLE KEY build = <ls_hri1028>-build
                        ASSIGNING <ls_t777a>.
    IF sy-subrc = 0.
      <ls_hri1028>-cname = <ls_t777a>-cname.
      <ls_hri1028>-stras = <ls_t777a>-stras.
      <ls_hri1028>-ort01 = <ls_t777a>-ort01.
      <ls_hri1028>-pstlz = <ls_t777a>-pstlz.
      <ls_hri1028>-land1 = <ls_t777a>-land1.
      <ls_hri1028>-regio = <ls_t777a>-regio.
      <ls_hri1028>-strs2 = <ls_t777a>-strs2.
      <ls_hri1028>-hausn = <ls_t777a>-hausn.

      IF <ls_hri1028>-telnr IS INITIAL.
        <ls_hri1028>-telnr = <ls_t777a>-telnr.
      ENDIF.

      IF <ls_hri1028>-faxnr IS INITIAL.
        <ls_hri1028>-faxnr = <ls_t777a>-faxnr.
      ENDIF.
    ENDIF.
  ENDLOOP.

ENDMETHOD.
