*  <--  p2        text
*----------------------------------------------------------------------*
FORM drill_down_rep USING row TYPE salv_de_row col TYPE salv_de_column.
  REFRESH :it_drill ,
           it_commit_drill,
           it_actfm_drill ,
           it_fi_drill .

  CASE col.
    WHEN 'YCOMMITMENT'.
      PERFORM ycommitment_drill  USING row col.

    WHEN 'YACTUAL_FM'.

      PERFORM yactual_fm_drill  USING row col.
    WHEN 'YACTUAL_FI'.
      PERFORM fi_drill  USING row col .

    WHEN 'YTOT_CONS'.
      PERFORM ytot_cons_drill  USING row col .

    WHEN OTHERS.

      READ TABLE it_final INTO DATA(ls_final) INDEX row.

      SELECT fm_area,docnr,docyear,docln,fiscyear,fundsctr,cmmtitem,
      budcat,process,budtype, lval01, lval02, lval03, lval04, lval05,
      lval06, lval07, lval08, lval09, lval10, lval11, lval12, lval13,
      lval14, lval15, lval16,text50
      FROM fmbl INTO TABLE @DATA(it_fmbl)" FOR ALL ENTRIES IN @it_final
      WHERE fm_area = @ls_final-fikrs AND fiscyear = @ls_final-yfyear AND
      budcat = '9F' AND fundsctr  = @ls_final-fictr AND cmmtitem = @ls_final-fipex .

  ENDCASE.

  LOOP AT it_fmbl INTO DATA(wa_fmbl) .
    wa_drill-fm_area = wa_fmbl-fm_area.
    wa_drill-docnr = wa_fmbl-docnr.
    wa_drill-docyear = wa_fmbl-docyear.
    wa_drill-docln = wa_fmbl-docln.
    wa_drill-fiscyear = wa_fmbl-fiscyear.
    wa_drill-fundsctr = wa_fmbl-fundsctr.
    wa_drill-cmmtitem = wa_fmbl-cmmtitem.
    wa_drill-process = wa_fmbl-process.
    wa_drill-budcat = wa_fmbl-budcat.
    wa_drill-budtype = wa_fmbl-budtype.
    wa_drill-text = wa_fmbl-text50.

    CASE  col .
      WHEN 'YBDG_OR' .
        IF  wa_fmbl-process  = 'ENTR' AND wa_fmbl-budtype =   'BE01' .
