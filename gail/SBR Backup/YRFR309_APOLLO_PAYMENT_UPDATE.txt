FORM get_data .

  SELECT a~gail_order_id, a~apollo_order_id, a~order_date, a~bill_no, a~bill_date, a~net_invoice_amt, a~apollo_gst_no,
           a~gail_gst_no, b~other_deduction, a~apollo_acc_no, a~utr_no, a~payment_amt, a~payment_date, a~apollo_bnk_name,
           a~ifsc_code, a~it_tds_amt, a~gst_tds_amt, b~deduction_disc
      FROM yrfa_cl_med_pi_h AS a
      INNER JOIN yrfa_cl_med_pi_i AS b
      ON a~gail_order_id = b~gail_order_id
      INTO TABLE @gi_alv
      WHERE a~payment_date IN @s_date.
*      GROUP BY a~gail_order_id, a~apollo_order_id, a~order_date, a~bill_no, a~bill_date.
  IF sy-subrc EQ 0.
    DATA(gi_data) = gi_alv[].
    SORT gi_alv BY gail_order_id.
    DELETE ADJACENT DUPLICATES FROM gi_alv.

    LOOP AT gi_alv ASSIGNING FIELD-SYMBOL(<lfs_alv>).

      "Other Deduction
      lv_oth_dis = REDUCE yy_other_disc( INIT x TYPE yy_other_disc FOR lw IN gi_data
                                         WHERE ( gail_order_id = <lfs_alv>-gail_order_id )
                                         NEXT x = x + lw-other_deduction ).
      <lfs_alv>-other_deduction = lv_oth_dis.

      "Deduction Discount
      lv_ded_dis = REDUCE yy_deduct_disc( INIT x TYPE yy_deduct_disc FOR lw IN gi_data
                                          WHERE ( gail_order_id = <lfs_alv>-gail_order_id )
                                          NEXT x = x + lw-deduction_disc ).
      <lfs_alv>-deduction_disc = lv_ded_dis.

    ENDLOOP.

  ENDIF.

ENDFORM.
