FORM validate_data .

  IF it_excel_data IS NOT INITIAL.

    REFRESH: it_kna1,it_lfa1,it_t176t.

    LOOP AT it_excel_data INTO wa_excel_data.
      CALL FUNCTION 'CONVERSION_EXIT_ALPHA_INPUT'
        EXPORTING
          input  = wa_excel_data-kunnr
        IMPORTING
          output = wa_excel_data-kunnr.

      CALL FUNCTION 'CONVERSION_EXIT_ALPHA_INPUT'
        EXPORTING
          input  = wa_excel_data-lifnr
        IMPORTING
          output = wa_excel_data-lifnr.

      MODIFY it_excel_data FROM wa_excel_data TRANSPORTING kunnr lifnr.
    ENDLOOP.

    SELECT kunnr FROM kna1 INTO CORRESPONDING FIELDS OF TABLE it_kna1
      FOR ALL ENTRIES IN it_excel_data
      WHERE kunnr EQ it_excel_data-kunnr.

    IF sy-subrc NE 0.
      MESSAGE : TEXT-003 TYPE 'E' DISPLAY LIKE 'E'.
    ENDIF.

    SELECT lifnr FROM lfa1 INTO CORRESPONDING FIELDS OF TABLE it_lfa1
      FOR ALL ENTRIES IN it_excel_data
      WHERE lifnr = it_excel_data-lifnr.

    IF sy-subrc NE 0.
      MESSAGE : TEXT-003 TYPE 'E' DISPLAY LIKE 'E'.
    ENDIF.

    SELECT bsark FROM t176t INTO CORRESPONDING FIELDS OF TABLE it_t176t
      FOR ALL ENTRIES IN it_excel_data
      WHERE bsark = it_excel_data-ygaid.

    IF sy-subrc NE 0.
      MESSAGE : TEXT-003 TYPE 'E' DISPLAY LIKE 'E'.
    ENDIF.
  ELSE.
    MESSAGE 'it_excel_data in INITIAL' TYPE 'E'.
  ENDIF.


  LOOP AT it_excel_data INTO wa_excel_data .

    IF  wa_excel_data-ydate GE sy-datum.
      CONCATENATE wa_excel_data-errmsg 'Data for current/ future date cannot be uploaded' INTO wa_excel_data-errmsg.
      wa_excel_data-error = 'X'.
    ENDIF.

    READ TABLE it_kna1 INTO DATA(tmp1) WITH KEY kunnr = wa_excel_data-kunnr  .
    IF sy-subrc NE 0.
      CONCATENATE wa_excel_data-errmsg 'Customer does not exist,' INTO wa_excel_data-errmsg.
      wa_excel_data-error = 'X'.
    ENDIF.

    READ TABLE it_lfa1 INTO DATA(tmp2) WITH KEY lifnr = wa_excel_data-lifnr.
    IF sy-subrc NE 0.
      CONCATENATE wa_excel_data-errmsg 'Vendor does not exist,' INTO wa_excel_data-errmsg.
      wa_excel_data-error = 'X'.
    ENDIF.

    READ TABLE it_t176t INTO DATA(tmp3) WITH KEY bsark = wa_excel_data-ygaid.
    IF sy-subrc NE 0.
      CONCATENATE wa_excel_data-errmsg 'GAID does not exist,' INTO wa_excel_data-errmsg.
      wa_excel_data-error = 'X'.
    ENDIF.
*
*    SELECT SINGLE * FROM T176T INTO @DATA(tmp3)
*      WHERE BSARK = @wa_excel_data-ygaid.
*    IF sy-subrc NE 0.
*      CONCATENATE wa_excel_data-errmsg 'GAID does not exist,' INTO wa_excel_data-errmsg.
*      wa_excel_data-error = 'X'.
*    ENDIF.

    IF wa_excel_data-ydate IS INITIAL.
      CONCATENATE wa_excel_data-errmsg 'DATE is null, ' INTO wa_excel_data-errmsg.
      wa_excel_data-error = 'X'.
    ENDIF.

    IF wa_excel_data-matnr IS INITIAL.
      CONCATENATE wa_excel_data-errmsg 'Material num is null, ' INTO wa_excel_data-errmsg.
      wa_excel_data-error = 'X'.
    ENDIF.

    IF wa_excel_data-lifnr IS INITIAL.
      CONCATENATE wa_excel_data-errmsg 'Vendor is null, ' INTO wa_excel_data-errmsg.
      wa_excel_data-error = 'X'.
    ENDIF.

    IF wa_excel_data-kunnr IS INITIAL.
      CONCATENATE wa_excel_data-errmsg 'Customer is null, ' INTO wa_excel_data-errmsg.
      wa_excel_data-error = 'X'.
    ENDIF.

    IF wa_excel_data-ygaid IS INITIAL.
      CONCATENATE wa_excel_data-errmsg 'GAIL ID is null, ' INTO wa_excel_data-errmsg.
      wa_excel_data-error = 'X'.
    ENDIF.

    IF wa_excel_data-rocode IS INITIAL.
      CONCATENATE wa_excel_data-errmsg 'RO Code is null, ' INTO wa_excel_data-errmsg.
      wa_excel_data-error = 'X'.
    ENDIF.
*** SOC BY ASHISH/SHREYOSI for CHARM 4000004497
*    IF wa_excel_data-kwmeng IS INITIAL.
*      CONCATENATE wa_excel_data-errmsg 'Measured Qty is null, ' INTO wa_excel_data-errmsg.
*      wa_excel_data-error = 'X'.
*    ENDIF.
*** EOC BY ASHISH/SHREYOSI for CHARM 4000004497
    IF wa_excel_data-msehi IS INITIAL.
      CONCATENATE wa_excel_data-errmsg 'UOM is null, ' INTO wa_excel_data-errmsg.
      wa_excel_data-error = 'X'.
    ENDIF.

    IF wa_excel_data-yystrm_sg IS INITIAL.
      CONCATENATE wa_excel_data-errmsg 'Density is null, ' INTO wa_excel_data-errmsg.
      wa_excel_data-error = 'X'.
    ENDIF.

    IF wa_excel_data-agcv IS INITIAL.
      CONCATENATE wa_excel_data-errmsg 'GCV is null, ' INTO wa_excel_data-errmsg.
      wa_excel_data-error = 'X'.
    ENDIF.

    IF wa_excel_data-ancv IS INITIAL.
      CONCATENATE wa_excel_data-errmsg 'NCV is null, ' INTO wa_excel_data-errmsg.
      wa_excel_data-error = 'X'.
    ENDIF.

    IF wa_excel_data-msehi CS 'SCM' AND wa_excel_data-ycqkg IS INITIAL.
      CLEAR var2.
      var2 = wa_excel_data-kwmeng * wa_excel_data-yystrm_sg.
      wa_excel_data-ycqkg = var2.
    ENDIF.

    IF wa_excel_data-msehi CS 'SCM' AND wa_excel_data-ycqkg IS NOT INITIAL.
      CLEAR var2.
      var2 = wa_excel_data-kwmeng * wa_excel_data-yystrm_sg.
      IF var2 NE wa_excel_data-ycqkg.
        CONCATENATE wa_excel_data-errmsg 'Invalid Qty-KG, ' INTO wa_excel_data-errmsg.
        wa_excel_data-error = 'X'.
      ENDIF.
      wa_excel_data-ycqkg = var2.
    ENDIF.

    IF wa_excel_data-msehi CS 'KG' AND wa_excel_data-ycqkg IS INITIAL.
      wa_excel_data-ycqkg = wa_excel_data-kwmeng.
    ENDIF.

    IF wa_excel_data-msehi CS 'KG' AND wa_excel_data-ycqkg IS NOT INITIAL.
      wa_excel_data-ycqkg = wa_excel_data-kwmeng.
    ENDIF.

    CLEAR var2.
    var2 = ( wa_excel_data-ycqkg / wa_excel_data-yystrm_sg ) * ( wa_excel_data-agcv / 1000000 ) * ( c_kcal ).

*** SOC BY ASHISH/SHREYOSI for CHARM 4000004497
*    IF wa_excel_data-ycqmbg IS INITIAL.
*      CONCATENATE wa_excel_data-errmsg 'Qty MBG is null, ' INTO wa_excel_data-errmsg.
*      wa_excel_data-error = 'X'.
    IF ( wa_excel_data-ycqmbg - var2 ) GE ( + c_err ).
      CONCATENATE wa_excel_data-errmsg 'Error in Qty-MBG, ' INTO wa_excel_data-errmsg.
      wa_excel_data-error = 'X'.
    ENDIF.
*** EOC BY ASHISH/SHREYOSI for CHARM 4000004497

    CLEAR var2.
    var2 = (  wa_excel_data-ycqkg / wa_excel_data-yystrm_sg ) * ( wa_excel_data-ancv / 1000000 ) * ( c_kcal ).

*** SOC BY ASHISH/SHREYOSI for CHARM 4000004497
*    IF wa_excel_data-ycqmbn IS INITIAL.
*      CONCATENATE wa_excel_data-errmsg 'Qty MBN is null, ' INTO wa_excel_data-errmsg.
*      wa_excel_data-error = 'X'.
*** EOC BY ASHISH/SHREYOSI for CHARM 4000004497

    IF ( wa_excel_data-ycqmbn - var2 ) GE ( + c_err ).
      CONCATENATE wa_excel_data-errmsg 'Error in Qty-MBN, ' INTO wa_excel_data-errmsg.
      wa_excel_data-error = 'X'.
    ENDIF.

    MODIFY it_excel_data FROM wa_excel_data TRANSPORTING error ycqkg.

    IF wa_excel_data-error NE 'X'.
      CLEAR  wa_tab_cbg.
      MOVE-CORRESPONDING wa_excel_data TO wa_tab_cbg.
      GET TIME STAMP FIELD wa_tab_cbg-timestamp.
      wa_tab_cbg-uname = sy-uname.
      APPEND wa_tab_cbg TO it_tab_cbg.
    ELSE.
      APPEND wa_excel_data TO it_excel_error.
    ENDIF.

    CLEAR wa_excel_data.
  ENDLOOP.
