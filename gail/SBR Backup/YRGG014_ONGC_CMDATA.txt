FORM SEND_BACKJOBDATA .
  SORT IT_ONGC BY YYBUS_LOCATION ASCENDING.
  IT_ONGC1[] = IT_ONGC[].
  DELETE IT_ONGC1 WHERE  YYBUS_LOCATION IS INITIAL.
  DELETE ADJACENT DUPLICATES FROM IT_ONGC1 COMPARING YY_ONGC_ID YYBUS_LOCATION YY_DATE.

  LOOP AT IT_ONGC1 INTO WA_ONGC1.
    LOOP AT IT_ONGC INTO WA_ONGC WHERE YY_ONGC_ID EQ WA_ONGC1-YY_ONGC_ID AND YYBUS_LOCATION EQ WA_ONGC1-YYBUS_LOCATION AND YY_DATE = WA_ONGC1-YY_DATE.
      WA_CMDATA-YYANMA_KEY    = WA_ONGC-YY_ONGC_ID.
      WA_CMDATA-YYBUS_LOCATION = WA_ONGC-YYBUS_LOCATION.
      WA_CMDATA-YYCUSTOMER = WA_ONGC-YYBUS_LOCATION.
      WA_CMDATA-YYDATE = WA_ONGC-YY_GAS_DAY.
      WA_CMDATA-YYSTRM_VOL_01 =  WA_CMDATA-YYSTRM_VOL_01 + WA_ONGC-YY_QUANTITY.
      WA_CMDATA-YYSTRM_GCV_01 = WA_CMDATA-YYSTRM_GCV_01 + ( WA_ONGC-YY_GCV * WA_ONGC-YY_QUANTITY ).
      WA_CMDATA-YYSTRM_NCV_01 = WA_CMDATA-YYSTRM_NCV_01 + ( WA_ONGC-YY_NCV * WA_ONGC-YY_QUANTITY ).
      WA_CMDATA-YYSTRMFLW_FLG01   = '+'.
      WA_CMDATA-YYSTRM_QUALITY01 = 'G'.
      WA_CMDATA-YYGCV_QUALITY01 = 'G'.
      WA_CMDATA-YYNCV_QUALITY01 = 'G'.
      WA_CMDATA-YYDATASOURCE = '07'.
    ENDLOOP.
    IF WA_CMDATA IS NOT INITIAL.
      WA_CMDATA-YYSTRM_GCV_01 = WA_CMDATA-YYSTRM_GCV_01 / WA_CMDATA-YYSTRM_VOL_01.
      WA_CMDATA-YYSTRM_NCV_01 = WA_CMDATA-YYSTRM_NCV_01 / WA_CMDATA-YYSTRM_VOL_01.
      WA_CMDATA-YYCUM_VOL = WA_CMDATA-YYSTRM_VOL_01.
      WA_CMDATA-YYAVG_GCV = WA_CMDATA-YYSTRM_GCV_01.
      WA_CMDATA-YYAVG_NCV = WA_CMDATA-YYSTRM_NCV_01.
      LV_VAL =   WA_CMDATA-YYAVG_GCV / WA_CMDATA-YYAVG_NCV .
      IF LV_VAL LT '1.06' OR LV_VAL GT '1.16'.
        WA_ONGC1-REMARKS = 'GCV/NCV IS OUT OF RANGE'.
        MODIFY IT_ONGC1 FROM WA_ONGC1 TRANSPORTING REMARKS.
        WA_CMDATA-YYCHECK = 'X'.
      ENDIF.
      WA_CMDATA-YYCDATE  = SY-DATUM.
      WA_CMDATA-YYTIME  = SY-UZEIT.
      WA_CMDATA-YYREVISEDBY  = SY-UNAME.
      CALL FUNCTION 'YRX_CHK_NG_TKT_STA'
        EXPORTING
          GAS_DAY        = WA_CMDATA-YYDATE
          LOCID          = WA_CMDATA-YYBUS_LOCATION
        IMPORTING
          INDICATOR      = INDICATOR
          MSG            = MSG
          PIPELINE       = PIPELINE
        TABLES
          ET_NOMINATIONS = ET_NOMINATIONS.

        WA_ONGC1-REMARKS = 'TICKET ALREADY EXIST'.
        MODIFY IT_ONGC1 FROM WA_ONGC1 TRANSPORTING REMARKS.
        WA_CMDATA-YYCHECK = 'X'.
      ENDIF.
      CONVERT DATE SY-DATUM TIME SY-UZEIT INTO TIME STAMP WA_CMDATA-YYTIMESTAMP TIME ZONE SY-ZONLO.
      APPEND WA_CMDATA TO IT_CMDATA.
    ENDIF.
    CLEAR WA_CMDATA.
    CLEAR: INDICATOR,MSG,PIPELINE.
    CLEAR WA_ONGC1.
  ENDLOOP.
  IF IT_CMDATA IS NOT INITIAL .
    LOOP AT IT_ONGC1 INTO DATA(FS_ONGC1).
      LOOP AT IT_ONGC INTO WA_ONGC WHERE YY_ONGC_ID EQ FS_ONGC1-YY_ONGC_ID AND  YYBUS_LOCATION EQ FS_ONGC1-YYBUS_LOCATION AND YY_DATE = FS_ONGC1-YY_DATE.
