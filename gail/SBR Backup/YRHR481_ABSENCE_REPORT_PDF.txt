FORM P_FETCH_VALID_PERNR.


  IF P_ENDDA LT P_BEGDA.
    MESSAGE 'ABSENCE END DATE CAN NOT BE GREATER THAN BEGIN DATE.' TYPE 'S' DISPLAY LIKE 'E'.
    LEAVE LIST-PROCESSING.
    EXIT.
  ENDIF.

  CLEAR: I_PERNR.                                     " FETCH VALID EMPLOYEES ON KEY DATE.
  IF P_BTRTL IS NOT INITIAL.
    SELECT B~PERNR B~BEGDA B~WERKS B~PERSK B~BTRTL B~PLANS B~ENAME  B~YY_DEPT  INTO TABLE I_PERNR
                    FROM PA0000 AS A INNER JOIN
                         PA0001 AS B ON A~PERNR EQ B~PERNR WHERE
                         B~PERNR IN P_PERNR AND B~BTRTL EQ P_BTRTL AND
                         B~PLANS NE '99999999' AND A~STAT2 EQ '3' AND
                         A~BEGDA LE P_ENDDA AND A~ENDDA GE P_ENDDA AND
                         B~BEGDA LE P_ENDDA AND B~ENDDA GE P_BEGDA.
  ELSE.
    SELECT B~PERNR B~BEGDA B~WERKS B~PERSK B~BTRTL B~PLANS B~ENAME  B~YY_DEPT  INTO TABLE I_PERNR
                    FROM PA0000 AS A INNER JOIN
                         PA0001 AS B ON A~PERNR EQ B~PERNR WHERE
                         B~PERNR IN P_PERNR AND
                         B~PLANS NE '99999999' AND A~STAT2 EQ '3' AND
                         A~BEGDA LE P_ENDDA AND A~ENDDA GE P_ENDDA AND
                         B~BEGDA LE P_ENDDA AND B~ENDDA GE P_BEGDA.
  ENDIF.
  IF SY-SUBRC EQ 0.

    SORT I_PERNR BY PERNR ASCENDING BEGDA DESCENDING.
    DELETE ADJACENT DUPLICATES FROM I_PERNR COMPARING PERNR.
    IF  RB1 IS NOT INITIAL.
      SELECT PERNR SUBTY BEGDA ENDDA BEGUZ ENDUZ FROM PA2001 INTO TABLE I_PA2001 FOR ALL ENTRIES IN I_PERNR
                                                             WHERE PERNR EQ I_PERNR-PERNR
                                                             AND AWART = '39'
                                                             AND BEGDA LE P_ENDDA
                                                             AND ENDDA GE P_BEGDA.
    ELSE.
      SELECT PERNR SUBTY BEGDA ENDDA BEGUZ ENDUZ FROM PA2001 INTO TABLE I_PA2001 FOR ALL ENTRIES IN I_PERNR
                                                             WHERE PERNR EQ I_PERNR-PERNR
                                                             AND AWART IN P_AWART
                                                             AND BEGDA LE P_ENDDA
                                                             AND ENDDA GE P_BEGDA.
    ENDIF.
    IF SY-SUBRC EQ 0.
      CLEAR: I_PA2001_TMP.
      SORT: I_PA2001.
      I_PA2001_TMP[] = I_PA2001[].
      SORT I_PA2001_TMP BY PERNR.
      DELETE ADJACENT DUPLICATES FROM I_PA2001_TMP COMPARING PERNR.
    ELSE.
      MESSAGE I004(YH01).
      LEAVE LIST-PROCESSING.
      EXIT.
    ENDIF.
  ELSE.
    MESSAGE I004(YH01).
    LEAVE LIST-PROCESSING.
    EXIT.
  ENDIF.

  IF RB2 = 'X'.
    PERFORM P_FETCH_LEAVE_40.
  ENDIF.

  IF RB3 = 'X'.
    PERFORM P_FETCH_LEAVE_CL_CHECK.
  ENDIF.

  IF RB4 = 'X'.
    PERFORM P_FETCH_INVALID_LEAVE_COMBO.
  ENDIF.

  IF RB5 = 'X'.
    PERFORM P_FETCH_INVALID_LEAVE_COMBO1.
  ENDIF.


  IF RB6 = 'X'.
    PERFORM P_FETCH_INVALID_LEAVE_COMBO2.
  ENDIF.

  IF RB7 = 'X'.
    PERFORM P_FETCH_INVALID_LEAVE_COMBO3.
  ENDIF.

ENDFORM.

FORM P_FETCH_LEAVES.
  TYPES: BEGIN OF TY_PTREQ_HEADER,
           REQUEST_ID    TYPE PTREQ_HEADER-REQUEST_ID,
           STATUS        TYPE PTREQ_HEADER-STATUS,
           MOD_USER      TYPE PTREQ_HEADER-MOD_USER,
           MOD_TIMESTAMP TYPE PTREQ_HEADER-MOD_TIMESTAMP,
         END OF TY_PTREQ_HEADER.

  DATA: L_BEGDA(10)   TYPE C,
        L_ENDDA(10)   TYPE C,
        I_LEAVES_TMP  TYPE TABLE OF TY_LEAVES,
        I_SENDER      TYPE TABLE OF TY_PTREQ_HEADER,
        I_APPROVER    TYPE TABLE OF TY_PTREQ_HEADER,
        WA_SENDER     TYPE TY_PTREQ_HEADER,
        WA_APPROVER   TYPE TY_PTREQ_HEADER,
        WA_LEAVES_TMP TYPE TY_LEAVES.

  CLEAR: I_T554T.
  SELECT * FROM T554T INTO TABLE I_T554T WHERE SPRSL = 'EN'.
  IF SY-SUBRC EQ 0.
    SORT I_T554T.
  ENDIF.

  CLEAR: I_T001P.
  SELECT WERKS BTRTL BTEXT FROM T001P INTO TABLE I_T001P FOR ALL ENTRIES IN I_PERNR WHERE
                                      WERKS EQ I_PERNR-WERKS AND BTRTL EQ I_PERNR-BTRTL.
  IF SY-SUBRC EQ 0.
    SORT I_T001P.
  ENDIF.

  CLEAR: IY_YRHA_DEPARTMENT.
  SELECT * FROM YRHA_DEPARTMENT INTO TABLE IY_YRHA_DEPARTMENT FOR ALL ENTRIES IN I_PERNR WHERE
                                                                  YY_DEPT EQ I_PERNR-YY_DEPT.
  IF SY-SUBRC EQ 0.
    SORT IY_YRHA_DEPARTMENT.
  ENDIF.

  CLEAR: I_HRP1000.
  SELECT * FROM HRP1000 INTO TABLE I_HRP1000 FOR ALL ENTRIES IN I_PERNR WHERE
                                 OBJID EQ I_PERNR-PLANS AND OTYPE EQ 'S' AND
                                 PLVAR EQ '01' AND BEGDA LE P_ENDDA AND ENDDA GE P_ENDDA.
  IF SY-SUBRC EQ 0.
    SORT I_HRP1000.
  ENDIF.

  CLEAR: I_LEAVES.
  SELECT A~REQUEST_ID C~PERNR C~SUBTY C~BEGDA C~ENDDA D~NOTICE_TEXT      " FOR LEAVE RESON TEXT.
     INTO CORRESPONDING FIELDS OF TABLE I_LEAVES FROM
     PTREQ_NOTICE     AS D INNER JOIN
     PTREQ_HEADER     AS A ON D~REQUEST_ID EQ A~REQUEST_ID INNER JOIN
     PTREQ_ITEMS      AS B ON A~ITEM_LIST_ID = B~ITEM_LIST_ID INNER JOIN
     PTREQ_ATTABSDATA AS C ON B~ITEM_INS = C~ITEM_ID
     FOR ALL ENTRIES IN I_PA2001_TMP WHERE C~PERNR = I_PA2001_TMP-PERNR AND
     D~VERSION_NO EQ 1 AND BEGDA LE P_ENDDA AND ENDDA GE P_BEGDA.
  IF SY-SUBRC EQ 0.
    SORT I_LEAVES BY REQUEST_ID ASCENDING PERNR ASCENDING.
  ENDIF.

  CLEAR: I_LEAVES_TMP.

        LT_PSP_PREV1  LIKE PDPSP   OCCURS 0 WITH HEADER LINE,
        LT_PSP_NEXT1  LIKE PDPSP   OCCURS 0 WITH HEADER LINE,
        LT_DAY_PSP1   LIKE PDSPPSP OCCURS 0 WITH HEADER LINE,
        LT_DAY_ATTR1  LIKE CASDAYATTR OCCURS 0 WITH HEADER LINE.


  IF  RB2 IS NOT INITIAL.
    SELECT PERNR , SUBTY , BEGDA , ENDDA , BEGUZ , ENDUZ
    FROM PA2001 INTO TABLE @DATA(I_PA2001_40)
                                WHERE PERNR IN @P_PERNR
                                AND AWART = '40'
                                AND BEGDA LE @P_ENDDA
                                AND ENDDA GE @P_BEGDA.

    IF I_PA2001_40 IS NOT INITIAL.
      IF P_BTRTL IS NOT INITIAL.
        SELECT PERNR, BEGDA, WERKS, PERSK, BTRTL ,PLANS, ENAME,  YY_DEPT FROM PA0001
          INTO TABLE @DATA(I_PERNR_PA0001_1)
          FOR ALL ENTRIES IN @I_PA2001_40
          WHERE PERNR = @I_PA2001_40-PERNR
          AND BTRTL = @P_BTRTL
          AND BEGDA LE @P_ENDDA AND ENDDA GE @P_ENDDA.

      ELSE.

        SELECT PERNR, BEGDA, WERKS, PERSK, BTRTL ,PLANS, ENAME,  YY_DEPT FROM PA0001
        INTO TABLE @DATA(I_PERNR_PA0001)
        FOR ALL ENTRIES IN @I_PA2001_40
        WHERE PERNR = @I_PA2001_40-PERNR
        AND BEGDA LE @P_ENDDA AND ENDDA GE @P_ENDDA.

      ENDIF.

    ENDIF.

    IF P_BTRTL IS NOT INITIAL.

      LOOP AT I_PA2001_40 INTO DATA(WA_40_1).
        READ TABLE I_PERNR_PA0001_1 INTO DATA(WA_BTLR) WITH KEY PERNR = WA_40_1-PERNR.
        IF SY-SUBRC = 4.
          DELETE I_PA2001_40 WHERE PERNR = WA_40_1-PERNR.
        ENDIF.

        CLEAR:WA_BTLR,WA_40_1.
      ENDLOOP.

    ENDIF.

    IF I_PERNR_PA0001_1 IS NOT INITIAL.
      I_PERNR_PA0001 = I_PERNR_PA0001_1.
    ENDIF.


    CLEAR: I_T554T.
    SELECT * FROM T554T INTO TABLE I_T554T WHERE SPRSL = 'EN'.

      SORT I_T554T.

FORM P_FETCH_ECO .
*** SOC BY SURAJ AND GAURAV FOR CHARM: 4000008578
  DATA: LT_PERNR_TAB TYPE STANDARD TABLE OF PDPNR,
        LT_PSP       TYPE STANDARD TABLE OF PDPSP,
        LT_DAY_PSP   TYPE STANDARD TABLE OF PDSPPSP,
        LW_PERNR_TAB TYPE PDPNR,
        LW_PSP       TYPE PDPSP,
        LW_DAY_PSP   TYPE PDSPPSP,
        LT_DAY_ATTR  TYPE  STANDARD TABLE OF CASDAYATTR..
  TYPES: BEGIN OF TY_WRKNG_SAT,
           PERNR TYPE PA0001-PERNR,
           WRKNG TYPE I,
         END OF TY_WRKNG_SAT.
  DATA: LT_WRKNG TYPE STANDARD TABLE OF TY_WRKNG_SAT,
        LT_LEAVE TYPE STANDARD TABLE OF TY_WRKNG_SAT,
        IT_LEAVE TYPE STANDARD TABLE OF TY_WRKNG_SAT,
        WA_LEAVE TYPE TY_WRKNG_SAT,
        LW_WRKNG TYPE TY_WRKNG_SAT.
  TYPES: BEGIN OF TY_SHIFT,
           PERNR TYPE PA0001-PERNR,
           GEN   TYPE I,
           REST  TYPE I,
           OFF   TYPE I,
           MRNG  TYPE I,
           EVNG  TYPE I,
           NIGHT TYPE I,
         END OF TY_SHIFT.
  DATA: LT_SHIFT TYPE STANDARD TABLE OF TY_SHIFT,
        WA_SHIFT TYPE TY_SHIFT.

  CLEAR: I_T554T.
  SELECT * FROM T554T INTO TABLE I_T554T WHERE SPRSL = 'EN'.
  IF SY-SUBRC EQ 0.
    SORT I_T554T.
  ENDIF.

  CLEAR: I_T001P.
  SELECT WERKS BTRTL BTEXT FROM T001P INTO TABLE I_T001P FOR ALL ENTRIES IN I_PERNR WHERE
                                      WERKS EQ I_PERNR-WERKS AND BTRTL EQ I_PERNR-BTRTL.
  IF SY-SUBRC EQ 0.
    SORT I_T001P.
  ENDIF.

  CLEAR: IY_YRHA_DEPARTMENT.
  SELECT * FROM YRHA_DEPARTMENT INTO TABLE IY_YRHA_DEPARTMENT FOR ALL ENTRIES IN I_PERNR WHERE
                                                                  YY_DEPT EQ I_PERNR-YY_DEPT.
  IF SY-SUBRC EQ 0.
    SORT IY_YRHA_DEPARTMENT.
  ENDIF.

  CLEAR: I_HRP1000.
  SELECT * FROM HRP1000 INTO TABLE I_HRP1000 FOR ALL ENTRIES IN I_PERNR WHERE
                                 OBJID EQ I_PERNR-PLANS AND OTYPE EQ 'S' AND
                                 PLVAR EQ '01' AND BEGDA LE P_ENDDA AND ENDDA GE P_ENDDA.
  IF SY-SUBRC EQ 0.
    SORT I_HRP1000.
  ENDIF.

  SELECT * FROM PA0015 INTO TABLE @DATA(I_PA0015)
                                  FOR ALL ENTRIES IN @I_PERNR
                                    WHERE PERNR = @I_PERNR-PERNR
                                      AND SUBTY = '1205'
                                      AND BEGDA GE @P_BEGDA
                                      AND ENDDA LE @P_ENDDA.
  IF I_PA0015 IS NOT INITIAL.
    DATA: LV_SUM TYPE ANZHL.
    DATA(LT_TEMP15) = I_PA0015[].
    SORT: LT_TEMP15 BY PERNR ASCENDING,
          I_PA0015 BY PERNR ASCENDING.
    DELETE ADJACENT DUPLICATES FROM LT_TEMP15 COMPARING PERNR.
    LOOP AT LT_TEMP15 ASSIGNING FIELD-SYMBOL(<FS_TEMP15>).
      CLEAR: LV_SUM.
      LOOP AT I_PA0015 INTO DATA(W_PA0015) WHERE PERNR = <FS_TEMP15>-PERNR.
        LV_SUM = LV_SUM + W_PA0015-ANZHL.
      ENDLOOP.
      <FS_TEMP15>-ANZHL = LV_SUM.
    ENDLOOP.
    CLEAR: I_PA0015.
    I_PA0015 = LT_TEMP15.
  ENDIF.


  CLEAR: LT_PSP, LT_DAY_PSP, LT_PERNR_TAB.
  LOOP AT I_PERNR INTO WA_PERNR.
    LW_PERNR_TAB-PERNR = WA_PERNR-PERNR.
    APPEND LW_PERNR_TAB TO LT_PERNR_TAB.
    CLEAR: LW_PERNR_TAB.
  ENDLOOP.
  CALL FUNCTION 'HR_PERSON_READ_WORK_SCHEDULE'
    EXPORTING
      BEGIN_DATE         = P_BEGDA
      END_DATE           = P_ENDDA
      READ_FROM_DATABASE = 'X'
    TABLES
      PERNR_TAB          = LT_PERNR_TAB
      PSP                = LT_PSP
      DAY_PSP            = LT_DAY_PSP.

  SORT LT_PSP BY PERNR DATUM ASCENDING.

  DATA(LT_PSP_TEMP) = LT_PSP[].
  DELETE LT_PSP WHERE TPROG  = 'GEN' AND FTKLA = '0'.

  LOOP AT LT_PSP_TEMP INTO LW_PSP.
    READ TABLE LT_PSP TRANSPORTING NO FIELDS WITH KEY DATUM = LW_PSP-DATUM PERNR = LW_PSP-PERNR.
    IF SY-SUBRC EQ 0.
      DELETE LT_PSP_TEMP WHERE DATUM =  LW_PSP-DATUM AND PERNR = LW_PSP-PERNR.
    ENDIF.
  ENDLOOP.

  LT_PSP = LT_PSP_TEMP.

  LOOP AT LT_PSP INTO LW_PSP.

    CALL FUNCTION 'DAY_ATTRIBUTES_GET'
      EXPORTING
        DATE_FROM                  = LW_PSP-DATUM
        DATE_TO                    = LW_PSP-DATUM
        LANGUAGE                   = SY-LANGU
      TABLES
        DAY_ATTRIBUTES             = LT_DAY_ATTR
      EXCEPTIONS
        FACTORY_CALENDAR_NOT_FOUND = 1
        HOLIDAY_CALENDAR_NOT_FOUND = 2
        DATE_HAS_INVALID_FORMAT    = 3
        DATE_INCONSISTENCY         = 4
        OTHERS                     = 5.

    READ TABLE LT_DAY_ATTR INTO DATA(LW_DAY_ATTR) INDEX 1.
    IF LW_DAY_ATTR-WEEKDAY NE '6'.
      DELETE LT_PSP WHERE DATUM = LW_PSP-DATUM.
    ENDIF.
    CLEAR: LT_DAY_ATTR, LW_DAY_ATTR .
  ENDLOOP.

  SORT LT_PSP BY PERNR.
  DATA(LT_PSP1) = LT_PSP[].
  DELETE ADJACENT DUPLICATES FROM LT_PSP1 COMPARING PERNR.
  LOOP AT LT_PSP1 INTO DATA(LW_PSP1).
    CLEAR: LW_WRKNG.
    LW_WRKNG-PERNR = LW_PSP1-PERNR.
    LOOP AT LT_PSP INTO LW_PSP WHERE PERNR = LW_PSP1-PERNR.
      LW_WRKNG-WRKNG = LW_WRKNG-WRKNG + 1.
    ENDLOOP.
    APPEND LW_WRKNG TO LT_WRKNG.
    CLEAR: LW_WRKNG.
  ENDLOOP.

  SELECT * FROM PA2001 INTO TABLE @DATA(LT_PA2001)
                                 WHERE PERNR IN @P_PERNR
*                                   AND BEGDA GE @P_BEGDA
                                   AND SUBTY NE '24'
                                   AND ENDDA GE @P_BEGDA
                                   AND BEGDA LE @P_ENDDA.

  SELECT * FROM PA2001 INTO TABLE @DATA(IT_PA2001)
                               WHERE PERNR IN @P_PERNR
                                 AND SUBTY IN ( '01', '02', '03', '04', '05', '07', '09',
                                                '10', '11', '12', '13', '14', '15', '16', '17', '18',
                                                '20', '21', '25', '26', '27', '28', '29', '30', '32',
                                                '33', '34', '35', '36', '37'  )
*                                 AND BEGDA GE @P_BEGDA
                                   AND ENDDA GE @P_BEGDA
                                   AND BEGDA LE @P_ENDDA.

  SELECT * FROM PA2003 INTO TABLE @DATA(IT_SHIFT)
                               WHERE PERNR IN @P_PERNR
*                                 AND BEGDA GE @P_BEGDA
                                   AND ENDDA BETWEEN @P_BEGDA AND @P_ENDDA.

  LOOP AT I_PERNR INTO WA_PERNR.
    READ TABLE IT_SHIFT INTO DATA(LW_SHIFT) WITH KEY PERNR = WA_PERNR-PERNR.
    IF SY-SUBRC <> 0.
      WA_SHIFT-PERNR =  WA_PERNR-PERNR.
      WA_SHIFT-GEN   =  P_ENDDA - P_BEGDA + 1.
      WA_SHIFT-REST  = 0.
      WA_SHIFT-OFF   = 0.
      WA_SHIFT-NIGHT = 0.
      WA_SHIFT-MRNG  = 0.
      WA_SHIFT-EVNG  = 0.
    ELSE.
      LOOP AT IT_SHIFT INTO LW_SHIFT WHERE PERNR = WA_PERNR-PERNR.
        WA_SHIFT-PERNR =  WA_PERNR-PERNR.
        IF LW_SHIFT-TPROG = 'REST'.
          WA_SHIFT-REST = WA_SHIFT-REST + LW_SHIFT-ENDDA - LW_SHIFT-BEGDA + 1.
        ELSEIF LW_SHIFT-TPROG = 'OFF'.
          WA_SHIFT-OFF = WA_SHIFT-OFF + LW_SHIFT-ENDDA - LW_SHIFT-BEGDA + 1.
        ELSEIF LW_SHIFT-TPROG = 'NGHT'.
          WA_SHIFT-NIGHT = WA_SHIFT-NIGHT + LW_SHIFT-ENDDA - LW_SHIFT-BEGDA + 1.
        ELSEIF LW_SHIFT-TPROG = 'EVNG'.
          WA_SHIFT-EVNG = WA_SHIFT-EVNG + LW_SHIFT-ENDDA - LW_SHIFT-BEGDA + 1.
        ELSEIF LW_SHIFT-TPROG = 'MRNG'.
          WA_SHIFT-MRNG = WA_SHIFT-MRNG + LW_SHIFT-ENDDA - LW_SHIFT-BEGDA + 1.
        ENDIF.
      ENDLOOP.
      WA_SHIFT-GEN = P_ENDDA - P_BEGDA + 1  -  WA_SHIFT-REST -  WA_SHIFT-OFF -  WA_SHIFT-NIGHT -  WA_SHIFT-MRNG -  WA_SHIFT-EVNG.
    ENDIF.
    APPEND WA_SHIFT TO LT_SHIFT.
    CLEAR: WA_SHIFT.
  ENDLOOP.


  DATA: LV_COUNTER TYPE I.
  SORT: LT_PA2001 BY PERNR,
        IT_PA2001 BY PERNR.
  CLEAR: LT_LEAVE, LV_COUNTER.
  DATA(LT_TEMP2001) = IT_PA2001[].
  DELETE ADJACENT DUPLICATES FROM LT_TEMP2001 COMPARING PERNR.
  LOOP AT LT_TEMP2001 INTO DATA(LW_TWMP2001).
    LOOP AT IT_PA2001 INTO DATA(WA_PA2001) WHERE PERNR = LW_TWMP2001-PERNR.
      IF WA_PA2001-BEGDA < P_BEGDA.
        WA_PA2001-BEGDA = P_BEGDA.
      ELSE.
        "DO NOTHING
      ENDIF.
      IF WA_PA2001-ENDDA > P_ENDDA.
        WA_PA2001-ENDDA = P_ENDDA.
      ELSE.
        " DO NOTHING
      ENDIF.
      LV_COUNTER = LV_COUNTER + WA_PA2001-ENDDA - WA_PA2001-BEGDA + 1.
    ENDLOOP.
    WA_LEAVE-PERNR = LW_TWMP2001-PERNR.
    WA_LEAVE-WRKNG = LV_COUNTER.
    APPEND WA_LEAVE TO IT_LEAVE.
    CLEAR: WA_LEAVE , LV_COUNTER.
  ENDLOOP.

  SORT: LT_PA2001 BY PERNR BEGDA ENDDA.
  DATA(LT_TEM2001) = LT_PA2001[].
  DATA(LT_COPY2001) = LT_PA2001[].
  CLEAR: LT_PA2001.
  DATA: LV_COUNTERX TYPE I.
  DELETE ADJACENT DUPLICATES FROM LT_TEM2001 COMPARING PERNR.
  LOOP AT LT_TEM2001 INTO DATA(LW_TEM2001).
    LOOP AT LT_COPY2001 INTO DATA(LW_COPY) WHERE PERNR = LW_TEM2001-PERNR.
      CLEAR: LV_COUNTERX.
      IF LW_COPY-BEGDA < P_BEGDA.
        LW_COPY-BEGDA = P_BEGDA.
      ELSE.
        "DO NOTHING
      ENDIF.
      IF LW_COPY-ENDDA > P_ENDDA.
        LW_COPY-ENDDA = P_ENDDA.
      ELSE.
        " DO NOTHING
      ENDIF.

      LV_COUNTERX = LW_COPY-ENDDA - LW_COPY-BEGDA + 1.
      DO LV_COUNTERX TIMES.
        LW_COPY-BEGDA = LW_COPY-BEGDA.
        LW_COPY-ENDDA = LW_COPY-BEGDA.
        APPEND LW_COPY TO LT_PA2001.
        LW_COPY-BEGDA = LW_COPY-BEGDA + 1.
      ENDDO.
    ENDLOOP.
  ENDLOOP.

  LOOP AT LT_PSP1 INTO LW_PSP1.
    CLEAR: LW_WRKNG.
    LW_WRKNG-PERNR = LW_PSP1-PERNR.
    LOOP AT LT_PSP INTO LW_PSP WHERE PERNR = LW_PSP1-PERNR.
      READ TABLE LT_PA2001 INTO DATA(LW_PA2001) WITH KEY PERNR = LW_PSP-PERNR BEGDA =  LW_PSP-DATUM.
      IF SY-SUBRC EQ 0.
*        IF LW_PSP-DATUM GE LW_PA2001-BEGDA AND LW_PSP-DATUM LE LW_PA2001-ENDDA.
        LW_WRKNG-WRKNG = LW_WRKNG-WRKNG + 1.
*        ENDIF.
      ENDIF.
    ENDLOOP.
    APPEND LW_WRKNG TO LT_LEAVE.
    CLEAR: LW_WRKNG.
  ENDLOOP.

  LOOP AT I_PERNR INTO WA_PERNR.
    W_ECO-PERNR = WA_PERNR-PERNR.
    W_ECO-GRADE = WA_PERNR-PERSK.
    W_ECO-ENAME = WA_PERNR-ENAME.

    CLEAR: WA_T001P.
    READ TABLE I_T001P INTO WA_T001P WITH KEY WERKS = WA_PERNR-WERKS BTRTL = WA_PERNR-BTRTL.
    IF SY-SUBRC EQ 0.
      W_ECO-LOCATION = WA_T001P-BTEXT.
    ENDIF.

    CLEAR: WA_HRP1000.
    READ TABLE I_HRP1000 INTO WA_HRP1000 WITH KEY OBJID = WA_PERNR-PLANS.
    IF SY-SUBRC EQ 0.
      W_ECO-DESIGNATION = WA_HRP1000-MC_STEXT.
    ENDIF.

    CLEAR: WA_YRHA_DEPARTMENT.
    READ TABLE IY_YRHA_DEPARTMENT INTO WA_YRHA_DEPARTMENT WITH KEY YY_DEPT = WA_PERNR-YY_DEPT.
    IF SY-SUBRC EQ 0.
      W_ECO-DEPARTMENT = WA_YRHA_DEPARTMENT-YY_DEPARTMENT.
    ENDIF.

    READ TABLE I_PA0015 INTO W_PA0015 WITH KEY PERNR = WA_PERNR-PERNR.
    IF SY-SUBRC EQ 0.
      W_ECO-YYECO_NO = W_PA0015-ANZHL.
    ENDIF.

    READ TABLE LT_WRKNG INTO LW_WRKNG WITH KEY PERNR = WA_PERNR-PERNR.
    IF SY-SUBRC EQ 0.
      W_ECO-WORK_SAT = LW_WRKNG-WRKNG.
    ENDIF.

    READ TABLE LT_LEAVE INTO DATA(LW_LEAVE) WITH KEY PERNR = WA_PERNR-PERNR.
    IF SY-SUBRC EQ 0.
      W_ECO-LEAVE_SAT = LW_LEAVE-WRKNG.
    ENDIF.

    READ TABLE IT_LEAVE INTO WA_LEAVE WITH KEY PERNR = WA_PERNR-PERNR.
    IF SY-SUBRC EQ 0.
      W_ECO-LEAVE_TOT = WA_LEAVE-WRKNG.
    ENDIF.

    READ TABLE LT_SHIFT INTO WA_SHIFT WITH KEY PERNR = WA_PERNR-PERNR.
    IF SY-SUBRC EQ 0.
      W_ECO-GEN = WA_SHIFT-GEN.
      W_ECO-REST = WA_SHIFT-REST.
      W_ECO-OFF = WA_SHIFT-OFF.
      W_ECO-MRNG = WA_SHIFT-MRNG.
      W_ECO-EVNG = WA_SHIFT-EVNG.
      W_ECO-NIGHT = WA_SHIFT-NIGHT.
    ENDIF.

    APPEND W_ECO TO I_ECO.
    CLEAR: W_ECO.
  ENDLOOP.
