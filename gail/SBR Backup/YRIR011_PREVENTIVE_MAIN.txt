  SELECT * INTO TABLE i_mpos
  FROM mpos

    WHERE iloan  = i_mpos-iloan
    AND   swerk = i_mpos-iwerk.
*    AND   TPLNR IN S_TPLNR.

    SELECT * INTO TABLE i_crhd
    FROM crhd
    FOR ALL ENTRIES IN i_mpos
    WHERE objid = i_mpos-gewrk
    AND   werks = i_mpos-iwerk
    AND   arbpl IN s_arbpl.

    SELECT * INTO TABLE i_t351x
    FROM t351x
    FOR ALL ENTRIES IN i_mpos
    WHERE spras = 'E'
    AND   strat = i_mpos-wstra.

  ENDIF.

  LOOP AT i_mpos INTO wa_mpos.
    READ TABLE i_iloa INTO wa_iloa WITH KEY iloan = wa_mpos-iloan.
    IF sy-subrc NE 0.
      READ TABLE i_crhd INTO wa_crhd WITH KEY objid = wa_mpos-gewrk.
      IF sy-subrc NE 0.
        DELETE i_mpos.
      ENDIF.
    ENDIF.
  ENDLOOP.

  IF i_mpos[] IS NOT INITIAL.
    SELECT * INTO TABLE i_mhis
    FROM mhis

    WHERE warpl = i_mpos-warpl
    AND   nplda IN s_nplda.
*    AND   ABRUD <> 0.

    FROM mpla
    FOR ALL ENTRIES IN i_mpos
    WHERE warpl = i_mpos-warpl.

    WHERE equnr = i_mpos-equnr.
  ENDIF.

    FROM mhio
    FOR ALL ENTRIES IN i_mhis

    AND   abnum = i_mhis-abnum.

    IF NOT i_mhio[] IS INITIAL.

      INTO TABLE i_afko
      FROM  afko
      FOR ALL ENTRIES IN i_mhio
      WHERE aufnr = i_mhio-aufnr.

      IF NOT i_afko[] IS INITIAL.

        INTO TABLE i_afvu
        FROM afvu
        FOR ALL ENTRIES IN i_afko WHERE
        aufpl = i_afko-aufpl AND

      ENDIF.
    ENDIF.

  ENDIF.
