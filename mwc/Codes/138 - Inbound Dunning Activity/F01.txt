FORM f_modify_screen .
  LOOP AT SCREEN .
    IF screen-group1 = TEXT-009.   "Target screen group
      IF rb3 <> abap_true.
        screen-active = 0.          "Hide/disable field
      ELSE.
        screen-active = 1.         "Show/enable field
      ENDIF.
      MODIFY SCREEN.               "Apply changes
    ENDIF.
    IF screen-group1 = 'MD2' OR screen-group1 = 'MD1'.
      screen-active = '0'.
      MODIFY SCREEN.
    ELSE.
      screen-active = '1'.
      MODIFY SCREEN.
    ENDIF.
    IF screen-name = 'S_SCP-LOW'.
      IF p_bukrs IS INITIAL.
        screen-input = '0'.  " disable field
      ELSE.
        screen-input = '1'.  " enable field
      ENDIF.
      MODIFY SCREEN.
    ENDIF.
  ENDLOOP.
ENDFORM.

  ELSEIF rb2 = abap_true.

    ELSE.

        ENDIF.
*** SECTION 2

*** SECTION 3 - LOOPING OVER THE FILENAMES

*& -->  p1        text
*& <--  p2        text

FORM f_get_viewdata .
  " Select records from custom table based on input filters
  SELECT company_code,service_provider,processed_on,filename, dateofupload, timeofupload,
         processed_by,num_of_records,last_acc_by,last_acc_on,error_file
    FROM zdt_m2c_dunfile
    WHERE company_code = @p_bukrs
      AND service_provider = @p_scp
      AND processed_on = @p_date
      AND outbound_file = @abap_false
      AND filename LIKE 'INDUN%'
      AND ( error_file = 'P' OR error_file = 'E' )
    INTO TABLE @DATA(lt_dunfile).
  IF sy-subrc = 0.
    " If records found, move to output table and display
    gt_output = CORRESPONDING #( lt_dunfile ).
    SORT gt_output BY company_code
                      service_provider
                      processed_on
                      dateofupload DESCENDING
                      timeofupload DESCENDING.
    PERFORM f_display_alv.
  ELSE.
    " Show message if no records found
    MESSAGE i082(zmc_m2c_fica) DISPLAY LIKE TEXT-e00.       " No files were created for given input
    LEAVE TO LIST-PROCESSING.
  ENDIF.


ENDFORM.

                      DATA(lv_mess) = |{ sy-msgv1 } { sy-msgv2 } { sy-msgv3 } { sy-msgv4 }|.
                      MESSAGE lv_mess TYPE 'I' DISPLAY LIKE 'E'.
                    ENDIF.
                    " Download binary data from internal table lt_srctab to frontend
                    CALL METHOD cl_gui_frontend_services=>gui_download
                      EXPORTING
                        bin_filesize            = lt_len
                        filename                = lv_filefullpath
                        filetype                = zcl_abap_utility=>zif_m2c_constant~zab_m2c_inbound_dunning-filetype_bin
                      CHANGING
                        data_tab                = lt_srctab
                      EXCEPTIONS
                        file_write_error        = 1
                        no_batch                = 2
                        gui_refuse_filetransfer = 3
                        invalid_type            = 4
                        no_authority            = 5
                        unknown_error           = 6
                        header_not_allowed      = 7
                        separator_not_allowed   = 8
                        filesize_not_allowed    = 9
                        header_too_long         = 10
                        dp_error_create         = 11
                        dp_error_send           = 12
                        dp_error_write          = 13
                        unknown_dp_error        = 14
                        access_denied           = 15
                        dp_out_of_memory        = 16
                        disk_full               = 17
                        dp_timeout              = 18
                        file_not_found          = 19
                        dataprovider_exception  = 20
                        control_flush_error     = 21
                        not_supported_by_gui    = 22
                        error_no_gui            = 23
                        OTHERS                  = 24.
                    IF sy-subrc <> 0.
                      MESSAGE: i063(zmc_m2c_fica) DISPLAY LIKE TEXT-e00.      " Selected file could not be downloaded
                    ENDIF.
                  ENDIF.
                ENDIF.
                " Clear temporary data
                CLEAR: lt_strings,lt_string_text.
              ELSE.
                MESSAGE i064(zmc_m2c_fica) DISPLAY LIKE TEXT-e00.
                LEAVE TO LIST-PROCESSING.
              ENDIF.
            ELSE.
              MESSAGE i065(zmc_m2c_fica) DISPLAY LIKE TEXT-e00.
              LEAVE TO LIST-PROCESSING.
            ENDIF.
          ENDIF.  " NO PATH FOUND IN TVARVC TABLE
        ENDIF.  " NO ENTRY FOUND IN DUNPATH TABLE
      ELSE.
        MESSAGE i066(zmc_m2c_fica) DISPLAY LIKE TEXT-e00.
        LEAVE TO LIST-PROCESSING.
      ENDIF.
    WHEN OTHERS.
  ENDCASE.

ENDFORM.
