*  REPLACE all OCCURRENCES OF 'BANSTARCBC' IN GV_FULLPATH WITH ''.

    IF SY-SYSID = 'QS4'.
     gv_archive_path =  '/BANKUPW12R2/H2H/QS4/REPORTING/Check_Payment_Posting/'.
    ENDIF.
    CONCATENATE gv_archive_path gv_file_path INTO gv_archive_path SEPARATED BY '/'.
  ENDIF.

  else.
    FIND REGEX '\\[^\\]*$' IN gv_fullpath MATCH OFFSET gv_pos    ##REGEX_POSIX.
*'\\[^\\]*$' TEXT-120
  IF sy-subrc = 0.
    gv_offset = gv_pos + 1.
    IF SY-sysid = 'QS4'.
    gv_archive_path = '/BANKUPW12R2/H2H/QS4/REPORTING/Check_Payment_Posting/'.
    ENDIF.
    gv_file_path = gv_fullpath+gv_offset.
    CONCATENATE gv_archive_path gv_file_path INTO gv_archive_path SEPARATED BY '/'.
  ENDIF.
  ENDIF.
*  FIND REGEX '\\[^\\]*$' IN gv_fullpath MATCH OFFSET gv_pos    ##REGEX_POSIX.

  REPLACE all OCCURRENCES OF 'BANSTARCBC' IN gv_archive_path WITH ''.
  REPLACE all OCCURRENCES OF 'BANSTASCB' IN gv_archive_path WITH ''.
  REPLACE all OCCURRENCES OF 'ACKRCBC' IN gv_archive_path WITH ''.
  REPLACE all OCCURRENCES OF 'ACKRCBC' IN gv_archive_path WITH ''.
  OPEN DATASET gv_archive_path FOR OUTPUT IN TEXT MODE ENCODING DEFAULT.

  IF gv_fullpath CS TEXT-002.
    "select data from acdoca table for hbkid and hktid
    IF gv_fullpath CS TEXT-003.
      IF gt_bansta IS NOT INITIAL.
        SELECT rbukrs, belnr, gjahr,hbkid, hktid,rwcur,wsl
        FROM acdoca
        INTO TABLE @DATA(lt_acdoca)
        FOR ALL ENTRIES IN @gt_bansta
        WHERE rbukrs = @gt_bansta-bukrs
        AND belnr = @gt_bansta-field1+0(10)
        AND gjahr = @gt_bansta-field4+0(4)
        AND rldnr = @gc_rldnr
        AND koart = @gc_koart
        AND hsl <> 0  .
      ENDIF.



    ELSEIF gv_fullpath CS TEXT-004.
      IF gt_bansta IS NOT INITIAL.
        "select data from acdoca table for hbkid and hktid
        SELECT rbukrs, belnr, gjahr,hbkid, hktid,rwcur,wsl
        FROM acdoca
        INTO TABLE @lt_acdoca
        FOR ALL ENTRIES IN @gt_bansta
        WHERE rbukrs = @gt_bansta-bukrs
        AND belnr = @gt_bansta-field1+0(10)
        AND gjahr = @gt_bansta-field6+0(4)
        AND rldnr = @gc_rldnr
        AND koart = @gc_koart
        AND  hsl  <> 0        .
      ENDIF.

    ENDIF.

    SORT lt_acdoca BY rbukrs belnr gjahr.
    "adding hbkid and hktid to the final table
    LOOP AT gt_bansta ASSIGNING <ls_bansta>.

FORM populate_data .
  CLEAR: gt_bansta, gt_ACK.

  LOOP AT gt_data INTO gs_data ##INTO_OK.
    SPLIT gs_data AT ',' INTO TABLE gt_fields.
    READ TABLE gt_fields INTO gs_fields INDEX 1.
    IF gs_fields = gc_h.
      LOOP AT gt_fields INTO gs_fields.
        REPLACE ALL OCCURRENCES OF '"' IN gs_fields WITH ''.
        gv_index = sy-tabix.
        ASSIGN COMPONENT gv_index OF STRUCTURE gs_bansta_hdr TO <fs_field>.
        IF sy-subrc = 0.
          <fs_field> = gs_fields.
        ENDIF.
      ENDLOOP.
      APPEND gs_bansta_hdr TO gt_bansta_hdr.
    ELSEIF gs_fields = gc_d ##BOOL_OK.
      LOOP AT gt_fields INTO gs_fields.
        REPLACE ALL OCCURRENCES OF '"' IN gs_fields WITH ''.
        gv_index = sy-tabix.
        ASSIGN COMPONENT gv_index OF STRUCTURE gs_bansta TO <fs_field>.
        IF sy-subrc = 0.
          IF gv_index = 2.
            <fs_field> = gs_fields+4.
            gs_bansta-bukrs = gs_fields+0(4).
          ELSE.
            <fs_field> = gs_fields.
          ENDIF.
        ENDIF.
      ENDLOOP.
      APPEND gs_bansta TO gt_bansta.
    ENDIF.
    CLEAR: gs_bansta, gs_bansta_hdr.
  ENDLOOP.
ENDFORM.

    CONCATENATE gv_archive_path gv_file_path INTO gv_archive_path SEPARATED BY '/'.
  ENDIF.

  lv_targetpath = gv_archive_path.

  CALL FUNCTION 'ARCHIVFILE_SERVER_TO_SERVER'
    EXPORTING
      sourcepath       = lv_sourcepath
      targetpath       = lv_targetpath
    IMPORTING
      length           = lv_length
    EXCEPTIONS
      error_file       = 1
      no_authorization = 2
      OTHERS           = 3.

  IF sy-subrc <> 0.
    MESSAGE ID sy-msgid TYPE sy-msgty NUMBER sy-msgno
               WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4.
  ENDIF.
ENDFORM.

*& <--  p2        text
*&---------------------------------------------------------------------*
FORM save_archive_file .
  DATA: lv_value TYPE string.
  CLEAR: gv_data.
  DO.
    ASSIGN COMPONENT sy-index OF STRUCTURE <fs_bansta> TO <fs_value>.
    IF sy-subrc <> 0.
      EXIT.
    ENDIF.

    lv_value = <fs_value>.
    CONCATENATE gv_data lv_value INTO gv_data SEPARATED BY ','.
  ENDDO.
  TRANSFER gv_data TO gv_archive_path.
ENDFORM.
