*-- Validate PO assignment type and project linkage
AT SELECTION-SCREEN ON p_ebeln.

  "Get account assignment category (P, Q, or N) from PO item
  SELECT SINGLE knttp
    FROM ekpo
    INTO @DATA(lv_knttp)
    WHERE ebeln = @p_ebeln
    AND ebelp = '10'.

  IF lv_knttp = 'P' OR lv_knttp = 'Q'.

    "Get WBS element from account assignment (EKKN)
    SELECT ps_psp_pnr
      FROM ekkn
      INTO  @DATA(lv_ekkn) UP TO 1 rows
      WHERE ebeln = @p_ebeln ORDER BY PRIMARY KEY. ENDSELECT.

    IF sy-subrc IS INITIAL.

      "Get order type from order master (AUFK) based on WBS
      SELECT auart
        FROM aufk
        INTO @DATA(lv_aufk) UP TO 1 rows
        WHERE pspel = @lv_ekkn ORDER BY PRIMARY KEY. ENDSELECT.

      IF sy-subrc IS INITIAL.

        "Validate if order type is valid
        IF lv_aufk <> 'ZMWC' AND lv_aufk <> 'ZNEZ'.
          DATA lv_msg TYPE string.
          CONCATENATE text-016 p_ebeln text-064 INTO lv_msg.
          MESSAGE: lv_msg TYPE 'E'.
        ENDIF.

      ENDIF.

    ENDIF.

  ELSEIF lv_knttp = 'N'.

    "Get network number for network-assigned PO
    SELECT nplnr
      FROM ekkn
      INTO @DATA(lv_ekkn1) UP TO 1 ROWS
      WHERE ebeln = @p_ebeln ORDER BY PRIMARY KEY. ENDSELECT.

    IF sy-subrc IS INITIAL.

      "Get order type from AUFK for network number
      SELECT auart
        FROM aufk
        INTO @lv_aufk UP TO 1 rows
        WHERE aufnr = @lv_ekkn1 ORDER BY PRIMARY KEY. ENDSELECT.

      IF sy-subrc IS INITIAL.

        "Validate order type for network-based PO
        IF lv_aufk <> 'ZMWC' AND lv_aufk <> 'ZNEZ'.
*          DATA lv_msg TYPE string.
          CONCATENATE text-016 p_ebeln text-064 INTO lv_msg.
          MESSAGE: lv_msg TYPE 'E'.

        ENDIF.

      ENDIF.

    ENDIF.

  ENDIF.
