FORM f_upload_customer_file .
*  DATA: f_fname TYPE string.
  DATA: t_lines TYPE I.
*  f_fname = fname.

  DATA: l_newko LIKE rf05a-newko.



*DATA: sr_no TYPE i VALUE 1.
*  FREE it_file.
  CALL FUNCTION 'TEXT_CONVERT_XLS_TO_SAP'
  EXPORTING
*     I_FIELD_SEPERATOR    =
*     I_LINE_HEADER        =
    i_tab_raw_data       = it_type
    i_filename           = fname
  TABLES
    i_tab_converted_data = t_customer
  EXCEPTIONS
    conversion_failed    = 1
    OTHERS               = 2.
  IF sy-subrc <> 0.
    MESSAGE ID sy-msgid TYPE sy-msgty NUMBER sy-msgno
    WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4.
  ENDIF.

  READ TABLE t_customer INDEX 1.
*  IF sy-subrc EQ 0.
*    MOVE-CORRESPONDING wa_file TO wa_head.
*    APPEND wa_head TO it_head.
*  ENDIF.
  DELETE t_customer  INDEX 1.
*  SR_NO = 0.
*  LOOP AT it_file INTO wa_file.
*    sr_no = sr_no + 1.
*    wa_file-sr_no =  sr_no .
*    wa_file-file_no = '1'.
*    MODIFY it_file FROM wa_file TRANSPORTING sr_no file_no.
*    CLEAR : wa_file.
*  ENDLOOP.

  LOOP AT t_customer.
    MOVE-CORRESPONDING t_customer TO t_cust.
    APPEND t_cust.
    CLEAR t_cust.

  ENDLOOP.


  LOOP AT t_cust.
    CLEAR: l_newko.

    IF t_cust-newko NA sy-abcde.
      g_length = STRLEN( t_cust-newko ).
      g_sublen = 10 - g_length.

      CASE g_sublen.
      WHEN '1'.
        CONCATENATE '0' t_cust-newko INTO l_newko.
        t_cust-newko = l_newko.
        MODIFY t_cust.
        CLEAR t_cust.

      WHEN '2'.
        CONCATENATE '00' t_cust-newko INTO l_newko.
        t_cust-newko = l_newko.
        MODIFY t_cust.
        CLEAR t_cust.

      WHEN '3'.
        CONCATENATE '000' t_cust-newko INTO l_newko.
        t_cust-newko = l_newko.
        MODIFY t_cust.
        CLEAR t_cust.

      WHEN '4'.
        CONCATENATE '0000' t_cust-newko INTO l_newko.
        t_cust-newko = l_newko.
        MODIFY t_cust.
        CLEAR t_cust.

      WHEN '5'.
        CONCATENATE '00000' t_cust-newko INTO l_newko.
        t_cust-newko = l_newko.
        MODIFY t_cust.
        CLEAR t_cust.

      WHEN '6'.
        CONCATENATE '000000' t_cust-newko INTO l_newko.
        t_cust-newko = l_newko.
        MODIFY t_cust.
        CLEAR t_cust.

      WHEN '7'.
        CONCATENATE '0000000' t_cust-newko INTO l_newko.
        t_cust-newko = l_newko.
        MODIFY t_cust.
        CLEAR t_cust.

      WHEN '8'.
        CONCATENATE '00000000' t_cust-newko INTO l_newko.
        t_cust-newko = l_newko.
        MODIFY t_cust.
        CLEAR t_cust.

      WHEN '9'.
        CONCATENATE '000000000' t_cust-newko INTO l_newko.
        t_cust-newko = l_newko.
        MODIFY t_cust.
        CLEAR t_cust.

      ENDCASE.

    ENDIF.

  ENDLOOP.


  SORT t_cust BY bktxt ASCENDING newbs DESCENDING.

  SELECT bschl koart FROM tbsl INTO TABLE t_tbsl
  FOR ALL ENTRIES IN t_cust WHERE bschl = t_cust-newbs.

  SORT t_cust BY newbs.
  SORT t_tbsl BY bschl.
  LOOP AT t_cust.
    READ TABLE t_tbsl WITH KEY bschl = t_cust-newbs BINARY SEARCH.
    IF sy-subrc = 0.
      t_cust-koart = t_tbsl-koart.
      MODIFY t_cust.
      CLEAR t_cust.
    ENDIF.
  ENDLOOP.


  "============="Start -  TDS===================
  SORT t_cust BY koart.


  IF tds2 NE 'X'.


    LOOP AT t_cust WHERE koart = 'K'.
      gt_lfbw-bukrs = t_cust-bukrs.
      gt_lfbw-lifnr = t_cust-newko.

      APPEND gt_lfbw.
      CLEAR gt_lfbw.
    ENDLOOP.

    SORT gt_lfbw BY bukrs lifnr.
    DELETE ADJACENT DUPLICATES FROM gt_lfbw.

    IF NOT gt_lfbw[] IS INITIAL.
      SORT gt_lfbw BY  bukrs lifnr.
      SELECT bukrs lifnr witht FROM lfbw INTO TABLE gt_lfbw1
      FOR ALL ENTRIES IN gt_lfbw
      WHERE bukrs EQ gt_lfbw-bukrs AND
      lifnr EQ gt_lfbw-lifnr AND
      wt_subjct EQ 'X' .
      SELECT * FROM t059p
      INTO TABLE gt_t059p
      FOR ALL ENTRIES IN gt_lfbw1
      WHERE land1 = 'IN'
      AND   witht = gt_lfbw1-witht
      AND wt_postm = '2'.

      LOOP AT gt_lfbw1.
        READ TABLE gt_t059p WITH KEY witht = gt_lfbw1-witht.
        IF sy-subrc = 0.
          gt_lfbw1-payment = 'X'.
        ELSE.
          gt_lfbw1-invoice = 'X'.
        ENDIF.
        MODIFY gt_lfbw1 TRANSPORTING payment invoice.
      ENDLOOP.


      SORT gt_lfbw BY  bukrs lifnr.
      SORT gt_lfbw1 BY  bukrs lifnr.

      DATA: ind TYPE sy-tabix.

      LOOP AT t_cust.
        ind = sy-tabix.

*        READ TABLE gt_lfbw1 WITH TABLE KEY lifnr = t_cust-newko
*                                           bukrs = t_cust-bukrs.
        IF sy-subrc = 0.
*          t_cust-tdsap = 'X'.
*          MODIFY t_cust INDEX ind TRANSPORTING tdsap.
*          CLEAR t_cust.
        ENDIF.

      ENDLOOP.

    ENDIF.

  ENDIF.






  LOOP AT t_cust WHERE koart = 'D'.
    gt_knbw-bukrs = t_cust-bukrs.
    gt_knbw-kunnr = t_cust-newko.

    APPEND gt_knbw.
    CLEAR gt_knbw.
  ENDLOOP.

  SORT gt_knbw BY bukrs kunnr.
  DELETE ADJACENT DUPLICATES FROM gt_knbw.

  IF NOT gt_knbw[] IS INITIAL.
    SORT gt_knbw BY  bukrs kunnr.
    SELECT bukrs kunnr FROM knbw INTO TABLE gt_knbw1
    FOR ALL ENTRIES IN gt_knbw
    WHERE bukrs EQ gt_knbw-bukrs AND
    kunnr EQ gt_knbw-kunnr AND
    wt_agent EQ 'X' .

    SORT gt_knbw BY  bukrs kunnr.
    SORT gt_knbw1 BY  bukrs kunnr.

    DATA: ind8 TYPE sy-tabix.

    LOOP AT t_cust.
      ind8 = sy-tabix.

      READ TABLE gt_knbw1 WITH TABLE KEY kunnr = t_cust-newko
      bukrs = t_cust-bukrs.
      IF sy-subrc = 0.
        t_cust-tdsap = 'X'.

        MODIFY t_cust INDEX ind8 TRANSPORTING tdsap.
        CLEAR t_cust.
      ENDIF.
    ENDLOOP.

  ENDIF.





  "=============END "TDS Logic"==============================


  "=============Start "TAX Logic"==============================


  LOOP AT t_cust WHERE koart EQ 'S' OR koart EQ 'A'.

    gt_skb1-bukrs = t_cust-bukrs.
    gt_skb1-saknr = t_cust-newko.
*
    APPEND gt_skb1.
    CLEAR gt_skb1.
  ENDLOOP.
  SORT gt_skb1 BY bukrs saknr.

  DELETE ADJACENT DUPLICATES FROM gt_skb1.

  IF NOT gt_skb1[] IS INITIAL.
    SORT gt_skb1 BY  bukrs saknr.

    SELECT bukrs saknr FROM skb1 INTO TABLE gt_skb1a FOR ALL ENTRIES IN gt_skb1
    WHERE bukrs EQ gt_skb1-bukrs
    AND saknr EQ gt_skb1-saknr
    AND mwskz IN ('>','<').

    SORT gt_skb1 BY  bukrs saknr.
    SORT gt_skb1a BY  bukrs saknr.

    DATA: ind1 TYPE sy-tabix.

    LOOP AT t_cust.
      ind1 = sy-tabix.

      READ TABLE gt_skb1a WITH TABLE KEY bukrs = t_cust-bukrs
      saknr = t_cust-newko.
      IF sy-subrc = 0.
        t_cust-taxap = 'X'.

        MODIFY t_cust INDEX ind1 TRANSPORTING taxap.
        CLEAR t_cust.
      ENDIF.
    ENDLOOP.

  ENDIF.

  "=============END "TAX Logic"==============================


  "=============START "CO Code currency Logic"==============================

  LOOP AT t_cust.

    gt_t001-bukrs = t_cust-bukrs.

    APPEND gt_t001.
    CLEAR gt_t001.
