FORM f009_bdc .
   DATA: lv_cur_begda TYPE t549q-begda,
        lv_cur_endda TYPE t549q-endda,
        lv_fol_begda TYPE t549q-begda,
        lv_fol_endda TYPE t549q-endda,
        lv_state     TYPE pv000-statetxt,
        lv_back_date TYPE sy-datum.

  LOOP AT it_filedata INTO wa_filedata.
    REFRESH: it_bdcdata[],it_messtab[].

    CALL FUNCTION 'CONVERSION_EXIT_ALPHA_INPUT'
      EXPORTING
        input  = wa_filedata-pernr
      IMPORTING
        output = v_pernr.

    SELECT SINGLE werks INTO v_werks
      FROM pa0001
      WHERE pernr = v_pernr
        AND endda = '99991231'.

    IF v_werks IS NOT INITIAL.
      CALL FUNCTION 'PA03_PCR_READ'
        EXPORTING
          f_abkrs               = wa_payroll_area-payroll_area
        IMPORTING
          f_current_begda       = lv_cur_begda
          f_current_endda       = lv_cur_endda
          f_following_begda     = lv_fol_begda
          f_following_endda     = lv_fol_endda
          f_state_text          = lv_state
        EXCEPTIONS
          abkrs_no_accounting   = 1
          pcr_does_not_exist    = 2
          abkrs_does_not_exist  = 3
          period_does_not_exist = 4
          no_pcr_authority      = 5
          OTHERS                = 6.
      IF sy-subrc = 0 AND lv_state IS NOT INITIAL.
        IF lv_state = 'Exit payroll'.
          IF wa_filedata-begda GE lv_fol_begda.
            CONCATENATE wa_filedata-begda+6(2) wa_filedata-begda+4(2) wa_filedata-begda+0(4) INTO wa_filedata-begda.
            CONCATENATE wa_filedata-endda+6(2) wa_filedata-endda+4(2) wa_filedata-endda+0(4) INTO wa_filedata-endda.
            CONCATENATE wa_filedata-dat01+6(2) wa_filedata-dat01+4(2) wa_filedata-dat01+0(4) INTO wa_filedata-dat01.

            PERFORM bdc_dynpro      USING 'SAPMP50A' '1000'.
            PERFORM bdc_field       USING 'BDC_OKCODE'
                                          '=INS'.
            PERFORM bdc_field       USING 'RP50G-PERNR'
                                          wa_filedata-pernr.
            PERFORM bdc_field       USING 'RP50G-TIMR6'
                                          'X'.
            PERFORM bdc_field       USING 'BDC_CURSOR'
                                          'RP50G-SUBTY'.
            PERFORM bdc_field       USING 'RP50G-CHOIC'
                                           '0591'.
            PERFORM bdc_field       USING 'RP50G-SUBTY'
                                          wa_filedata-subty.
            PERFORM bdc_dynpro      USING 'MP059100' '2000'.
            PERFORM bdc_field       USING 'BDC_CURSOR'
                                          'ITAB_NOM-GRADD(01)'.
            PERFORM bdc_field       USING 'BDC_OKCODE'
                                          'UPD'.
            PERFORM bdc_field       USING 'P0591-BEGDA'
                                          wa_filedata-begda.
            PERFORM bdc_field       USING 'P0591-ENDDA'
                                          wa_filedata-endda.

FORM f_write_message .
  DATA : lv_msg TYPE string.
  IF wa_filedata IS NOT INITIAL.

    IF     it_messtab[] IS NOT INITIAL.
      LOOP AT it_messtab.
        CLEAR lv_msg.
        CALL FUNCTION 'FORMAT_MESSAGE'
          EXPORTING
            id        = it_messtab-msgid
            lang      = sy-langu
            no        = it_messtab-msgnr
            v1        = it_messtab-msgv1
            v2        = it_messtab-msgv2
            v3        = it_messtab-msgv3
            v4        = it_messtab-msgv4
          IMPORTING
            msg       = lv_msg
          EXCEPTIONS
            not_found = 1
            OTHERS    = 2.
*      IF sy-subrc <> 0.
*        MESSAGE ID sy-msgid TYPE sy-msgty NUMBER sy-msgno
*                WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4.
*      ENDIF.
*      WRITE :/ lv_msg.

          wa_filedata-zindicator = 'S'.
          wa_mail-msgtyp = wa_filedata-zindicator.
          wa_mail-msg = 'Employee successfully hired.'.
        ENDIF.

          IF it_messtab-msgid = 'PG' AND it_messtab-msgnr = '001'.
            wa_filedata-zindicator = 'E'.

            wa_filedata-zindicator = it_messtab-msgtyp.
          ENDIF.
          wa_mail-msgtyp = wa_filedata-zindicator.
*        wa_mail-msg = 'Employee successfully hired.'.
          wa_mail-msg = lv_msg.
        ENDIF.
        IF it_messtab-msgtyp = c_e .
          wa_filedata-zindicator = it_messtab-msgtyp.
          wa_mail-msgtyp = wa_filedata-zindicator.
          wa_mail-msg = lv_msg.
        ENDIF.
      ENDLOOP.

      SELECT infty subty
      INTO CORRESPONDING FIELDS OF TABLE it_t591a
      FROM t591a
      WHERE infty = '0591'
      AND   subty = wa_filedata-subty.

        wa_filedata-zindicator = 'E'.
        wa_mail-msgtyp = wa_filedata-zindicator.
        wa_mail-msg = 'Invalid subtype'.
      ENDIF.


      IF wa_filedata-pernr EQ ' ' .
        wa_filedata-zindicator = 'E'.
        wa_mail-msgtyp = wa_filedata-zindicator.

*FTP_R3_TO_SERVER:used to transfer the internal table data as a file to other system in the character mode.
  CALL FUNCTION 'FTP_R3_TO_SERVER'
    EXPORTING
      handle         = w_hdl
      fname          = wrk_file          "file path of destination system
*     fname          = 'ftp://astralsftp.astralpipes.com:4443/New%20Folder/SAP/cust.text'          "file path of destination system
      character_mode = 'X'
    TABLES
      text           = it_dat1
    EXCEPTIONS
      tcpip_error    = 1
      command_error  = 2
      data_error     = 3
      OTHERS         = 4.
  IF sy-subrc <> 0.
    MESSAGE ID sy-msgid TYPE sy-msgty NUMBER sy-msgno
    WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4
    RAISING invalid_output_file.
  ENDIF.
*FTP_DISCONNECT: This is used to disconnect the connection between SAP and other system.
* To disconnect the FTP
  CALL FUNCTION 'FTP_DISCONNECT'
    EXPORTING
      handle = w_hdl.
*RFC_CONNECTION_CLOSE:This is used to disconnect the RFC connection between SAP and other system.
  CALL FUNCTION 'RFC_CONNECTION_CLOSE'
    EXPORTING
      destination = l_dest
    EXCEPTIONS
      OTHERS      = 1.
ENDFORM.
