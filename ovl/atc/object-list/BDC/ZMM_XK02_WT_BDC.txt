FORM bdc_dynpro USING program screen.

  wa_bdc-program = program.
  wa_bdc-dynpro = screen.
  wa_bdc-dynbegin = 'X'.
  APPEND wa_bdc TO it_bdc.
  CLEAR:wa_bdc.
ENDFORM.

  ENDIF.

ENDFORM.

*----------------------------------------------------------------------*

*----------------------------------------------------------------------*
FORM authorization_check .
  DATA: lv_bukrs TYPE bukrs.
  CLEAR: lv_bukrs.
  LOOP AT lt_xk02 INTO DATA(wa_data).
    lv_bukrs = wa_data-company_code.
    AUTHORITY-CHECK OBJECT 'ZXK02_WT'
        ID 'BUKRS' FIELD lv_bukrs
        ID 'ACTVT' FIELD '02'.
    IF sy-subrc <> 0.
      MESSAGE e026(zfi) WITH lv_bukrs.
    ENDIF.
    CLEAR : wa_data,lv_bukrs.
  ENDLOOP.

ENDFORM.
