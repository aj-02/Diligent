FORM final_file_for_bapi .



  CLEAR: wa_final, counter.
  counter = '900000001'.
  CLEAR wa_final.
  LOOP AT it_final INTO wa_final.
    v_srno = wa_final-sr_no.
    v_fno  = wa_final-file_no.
    CONDENSE: v_srno, v_fno.

    CONCATENATE v_fno v_srno INTO v_srno.
    wa_final-sr_no = v_srno.
    MODIFY it_final FROM wa_final TRANSPORTING sr_no.

    IF wa_final-newko CS 'L'.
      IF wa_final-newbs EQ '50'.
        wa_final-wrbtr = wa_final-wrbtr * -1.
        wa_final-dmbtr = wa_final-dmbtr * -1.
        wa_final-dmbe2 = wa_final-dmbe2 * -1.
        wa_final-kostl = ''.
        wa_final-shkzg = 'H'.
        wa_final-koart = 'S'.

        MODIFY it_final FROM wa_final TRANSPORTING wrbtr dmbtr dmbe2 kostl shkzg koart.
    ELSEIF wa_final-newbs EQ '40'.
        wa_final-shkzg = 'S'.
        wa_final-koart = 'S'.
        MODIFY it_final FROM wa_final TRANSPORTING shkzg koart.
      ENDIF.
      wa_final-sr_no = counter.
      MODIFY it_final FROM wa_final TRANSPORTING sr_no.
      counter = counter + 1.

    ENDIF.


    IF wa_final-shkzg EQ 'H' AND wa_final-newko NS 'L'.
      wa_final-wrbtr = wa_final-wrbtr * -1.
      wa_final-dmbtr = wa_final-dmbtr * -1.
      wa_final-dmbe2 = wa_final-dmbe2 * -1.
      MODIFY it_final FROM wa_final TRANSPORTING wrbtr dmbtr dmbe2.
    ENDIF.
    CLEAR: v_srno, v_fno.

    IF MAX NE 1.
      CLEAR: wa_final-kursf, wa_final-xblnr.
      MODIFY it_final FROM wa_final TRANSPORTING kursf xblnr.
    ENDIF.



    CLEAR wa_j_1iexcacc.
    READ TABLE it_j_1iexcacc INTO wa_j_1iexcacc WITH KEY j_1ibukrs = wa_final-bukrs
    j_1ionhold = wa_final-newko.
    IF sy-subrc = 0.
      IF wa_final-koart = 'S'.
*        APPEND wa_file TO it_file_j_1iexcacc.
*        DELETE it_file_general WHERE bukrs EQ wa_file-bukrs
*                                 AND newko EQ wa_file-newko
*                                 AND koart = 'S'.
        CLEAR wa_final-ldgrp.
        MODIFY it_final FROM wa_final TRANSPORTING ldgrp.
      ENDIF.
    ENDIF.
  ENDLOOP.
  CLEAR wa_final.

FORM bapi_check .
  CALL FUNCTION 'BAPI_ACC_DOCUMENT_CHECK'
  EXPORTING
    documentheader    = gs_documentheader
  TABLES
    accountgl         = gt_accountgl
    accountreceivable = gt_accountreceivable
    accountpayable    = gt_accountpayable
    accounttax        = gt_accounttax
    currencyamount    = gt_currencyamount
    RETURN            = gt_return
    extension2        = gt_extension2
    accountwt         = gt_accountwt.
  IF gt_return IS NOT INITIAL.
    CLEAR wa_return.
    READ TABLE gt_return INTO wa_return INDEX 1.
    IF wa_return-ID = 'RW' AND wa_return-NUMBER = '614'.
    ELSE.

      IF oa = 'X'.
        LOOP AT gt_return INTO wa_return.

          asset_gl = wa_return-message_v1.

          CALL FUNCTION 'CONVERSION_EXIT_ALPHA_INPUT'
          EXPORTING
            INPUT  = asset_gl
          IMPORTING
            OUTPUT = asset_gl.

          wa_return-message_v1 = asset_gl.
          MODIFY gt_return FROM wa_return TRANSPORTING message_v1.
        ENDLOOP.

        LOOP AT gt_return INTO wa_return.
          READ TABLE it_file_asset1 INTO wa_file_asset1
          WITH KEY  newko = wa_return-message_v1
          bukrs = wa_return-message_v2.
          IF sy-subrc = 0.
            IF wa_return-ID = 'F5' AND wa_return-NUMBER = '354'.

              wa_return-FIELD = 'ASSET' .
              MODIFY gt_return FROM wa_return TRANSPORTING FIELD.
            ENDIF.
          ENDIF.
        ENDLOOP.
      ENDIF.


      LOOP AT gt_return INTO wa_return.
        IF wa_return-FIELD <> 'ASSET' AND ( wa_return-ID <> 'RW' AND  wa_return-NUMBER <> '609' ).

          APPEND wa_return TO gt_return1.
        ENDIF.
      ENDLOOP.
    ENDIF.

  ENDIF.
  REFRESH : gt_accountgl[],
  gt_accountreceivable[],
  gt_accountpayable[],
  gt_currencyamount[],
  gt_return[],
  gt_extension2[],

  gt_accounttax[].
