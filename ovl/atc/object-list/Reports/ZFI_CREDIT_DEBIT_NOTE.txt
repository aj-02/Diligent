      CLEAR : wa_header,v_cr,v_wrbtr,v_plant,wa_skat,v_lifnr.
    ENDAT.
    CLEAR : wa_bseg,wa_sgtxt,wa_bkpf,wa_item,wa_bsak,v_bukrs.
  ENDLOOP.

ENDFORM.

FORM display_smartform_zp .
  CALL FUNCTION 'SSF_FUNCTION_MODULE_NAME'
    EXPORTING
      formname           = 'ZFI_BANK_RECEIPT'
    IMPORTING
      fm_name            = fname
    EXCEPTIONS
      no_form            = 1
      no_function_module = 2
      OTHERS             = 3.
  IF sy-subrc <> 0.
    MESSAGE 'WRONG SMARTFORM NAME' TYPE 'E'.
  ENDIF.

  DESCRIBE TABLE it_header LINES w_cnt.

  LOOP AT it_header INTO wa_header.
    IF wa_header-blart = 'KZ' OR wa_header-blart = 'ZP'.
      wa_texts-head = 'Bank Payment Voucher'.
    ELSE.
      wa_texts-head = 'Bank Receipt Voucher'.
    ENDIF.
*    wa_texts-number = 'Voucher No'.
*    wa_texts-date = 'Voucher Date'.

    wa_texts-number = v_vno. " Added by Carina Jose on 08.04.2022
    wa_texts-date =  v_vdt. " Added by Carina Jose on 08.04.2022
    CONDENSE wa_texts-head.
    CONDENSE wa_texts-number.
    CONDENSE wa_texts-date.

    LOOP AT it_item INTO wa_item WHERE augbl = wa_header-belnr.
*IF WA_ITEM-AUGBL NE WA_ITEM-BELNR.
*IF WA_ITEM-AUGBL NE WA_ITEM-BELNR.
      MOVE-CORRESPONDING wa_item TO wa_item1.
      APPEND wa_item1 TO it_item1.
      CLEAR : wa_item,wa_item1.
*ELSE.
*MOVE-CORRESPONDING WA_ITEM TO WA_NARR.
*APPEND WA_NARR TO IT_NARR.
*CLEAR : WA_ITEM,WA_NARR.
*ENDIF.
    ENDLOOP.

    IF wa_header-kursf IS INITIAL.
      wa_header-kursf = 1.
    ENDIF.

    IF w_cnt GT 1.
      w_cnt2 = sy-tabix .
      CASE w_cnt2.
        WHEN 1.
          control_parameters-no_open   = space .
          control_parameters-no_close  = 'X' .
        WHEN w_cnt .
          control_parameters-no_open   = 'X' .
          control_parameters-no_close  = space .
        WHEN OTHERS.
          control_parameters-no_open   = 'X' .
          control_parameters-no_close  = 'X' .
      ENDCASE.
    ENDIF.

    IF w_cnt = 1.
      CALL FUNCTION fname
        EXPORTING
          wa_header        = wa_header
          it_item          = it_item1
          wa_texts         = wa_texts
          it_narr          = it_narr
          v_fcompname      = v_fcompname
          v_butxt          = v_butxt
        EXCEPTIONS
          formatting_error = 1
          internal_error   = 2
          send_error       = 3
          user_canceled    = 4
          OTHERS           = 5.
      IF sy-subrc <> 0.
      ENDIF.
    ENDIF.

    IF w_cnt GT 1.
      CALL FUNCTION fname
        EXPORTING
          control_parameters = control_parameters
          wa_header          = wa_header
          it_item            = it_item1
          wa_texts           = wa_texts
          it_narr            = it_narr
          v_fcompname        = v_fcompname
          v_butxt            = v_butxt
        EXCEPTIONS
          formatting_error   = 1
          internal_error     = 2
          send_error         = 3
          user_canceled      = 4
          OTHERS             = 5.
      IF sy-subrc <> 0.
      ENDIF.
    ENDIF.


    CLEAR : wa_header,wa_texts,p1,v_waerk,v_lifnr.
    REFRESH : it_item1.
  ENDLOOP.

ENDFORM.
