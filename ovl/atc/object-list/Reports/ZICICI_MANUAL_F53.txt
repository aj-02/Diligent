FORM moving_files.
  DATA: v_file_count  TYPE i,
        v_dir_count   TYPE i,
        it_file_table LIKE sdokpath OCCURS 0 WITH HEADER LINE,
        it_dir_table  LIKE sdokpath OCCURS 0 WITH HEADER LINE,
        v_return      TYPE i.

  CONSTANTS : c_import(50) VALUE 'C:\ICICI\IMPORT\',
              c_filter(4)  VALUE '*.*'.

  CALL FUNCTION 'TMP_GUI_DIRECTORY_LIST_FILES'
    EXPORTING
      directory  = c_import
      filter     = c_filter
    IMPORTING
      file_count = v_file_count
      dir_count  = v_dir_count
    TABLES
      file_table = it_file_table
      dir_table  = it_dir_table
    EXCEPTIONS
      cntl_error = 1
      OTHERS     = 2.

  IF sy-subrc = 0.
    LOOP AT it_file_table.
      CLEAR: v_source, v_destination.
      CONCATENATE c_source  it_file_table-pathname
             INTO v_source.
      CONCATENATE c_desti   it_file_table-pathname
             INTO v_destination.

      CALL FUNCTION 'WS_FILE_COPY'
        EXPORTING
          destination = v_destination
          source      = v_source
        IMPORTING
          return      = v_return.

      CALL FUNCTION 'WS_FILE_DELETE'
        EXPORTING
          file   = v_source
        IMPORTING
          return = v_return.
    ENDLOOP.

    CALL FUNCTION 'TMP_GUI_DIRECTORY_LIST_FILES'
      EXPORTING
        directory  = v_source
        filter     = c_filter
      IMPORTING
        file_count = v_file_count
        dir_count  = v_dir_count
      TABLES
        file_table = it_file_table
        dir_table  = it_dir_table
      EXCEPTIONS
        cntl_error = 1
        OTHERS     = 2.
    IF v_file_count > 0.
      CLEAR lv_message.

      CONCATENATE 'Error copying files from C:\ICICI\IMPORT'
                  'to C:\ICICI\PROCESSED folder'
                  INTO lv_message SEPARATED BY space.
      alv_pop1     c_error
                 lv_message
                 ''
                 ''
                 ''
                 ''.

      alv_pop2     ''
                   ''
                   ''
                   ''
                   ''
                   ''
                   ''.
    ENDIF.
  ENDIF.

ENDFORM.                    "moving_files

FORM payfile_format.
  FIELD-SYMBOLS: <field>    TYPE any, <struc> TYPE any,
                 <invfield> TYPE any, <invstruc> TYPE any.
  DATA m_structure(20) VALUE 'H1'.
  ASSIGN (m_structure) TO <struc>.
  CLEAR x_data.
  DO.
    ASSIGN COMPONENT sy-index OF STRUCTURE <struc> TO <field>.
    IF sy-subrc <> 0.
      EXIT.
    ENDIF.
    CONCATENATE x_data  <field> '|' INTO x_data.
    ""  CONCATENATE '|' X_DATA  <FIELD>  INTO X_DATA.

*      REPLACE ALL OCCURRENCES OF SUBSTRING '|' IN X_DATA WITH ''.
  ENDDO.
  len1 = strlen( x_data ).
  SUBTRACT 1 FROM len1.

*    CLEAR full.
*    FULL = X_DATA.
*    CLEAR x_data.
*    shift FULL RIGHT DELETING TRAILING '|'.
*    x_data = full.
  "" SHIFT X_DATA by 1 PLACES.
  APPEND x_data+0(len1) TO it_data.
  ""CLEAR  full.
  CLEAR x_data.
  CLEAR len1.

  IF it_iciciadvice[] IS NOT INITIAL.
    CLEAR x_data.
    LOOP AT it_iciciadvice ASSIGNING <invstruc>.
      DO.
        ASSIGN COMPONENT sy-index OF STRUCTURE <invstruc> TO <invfield>.
        IF sy-subrc <> 0.
          EXIT.
        ENDIF.
        CONCATENATE x_data  <invfield> '|' INTO x_data.
      ENDDO.
      len1 = strlen( x_data ).
      SUBTRACT 1 FROM len1.
      APPEND x_data+0(len1) TO it_data.

      CLEAR x_data.
      CLEAR len1.
    ENDLOOP.
    CLEAR: <invstruc>.
  ENDIF.
ENDFORM.

  DATA: m_char       TYPE c,

        v_pos    TYPE i,
        v_strlen TYPE i.

          SET PARAMETER ID 'BUK' FIELD s_bukrs.

          SET PARAMETER ID 'BUK' FIELD s_bukrs.
