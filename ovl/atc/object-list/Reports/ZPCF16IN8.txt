*  <--  p2        text
*----------------------------------------------------------------------*
FORM GET_EXCEL_DATA.
  DATA: p_year type t549q-pabrj,
        p_period type t549q-pabrp,
        p_date type t549q-pabrp,
        PYDTE TYPE t549q-begda.
  DATA : wa_final  LIKE final_tab,
         tmp_ctnr2 LIKE final_tab-cntr2.
  CLEAR tmp_ctnr2.
  LOOP AT EMP_TAB.
  loop at excel_tab where emp_no eq emp_tab-emp_no .           "RK1067021
    clear TEMSEFIN_TAB.
    IF MULTIPLE_F16 = 'X'.
      IF  COMP_CD NE EXCEL_TAB-BUKRS.
        CONTINUE.
      ENDIF.
    ENDIF.
      CLEAR wa_final.
    PERFORM CONV_DATE USING EXCEL_TAB-DT_CHALLAN CHANGING N_DATE.
      PERFORM CONV_DATE USING EXCEL_TAB-BEGDA      CHANGING PAY_BEGDA.
      PERFORM CONV_DATE USING EXCEL_TAB-ENDDA      CHANGING PAY_ENDDA.
      LOOP AT FINAL_TAB into WA_FINAL WHERE F16_BEGDA LE PAY_BEGDA
                                        AND F16_ENDDA GE PAY_ENDDA
                                        AND PERNR     EQ EMP_TAB-EMP_NO.
          TEMSEFIN_TAB-CNTR2 = WA_FINAL-CNTR2.
                   TMP_CTNR2 = WA_FINAL-CNTR2.
      ENDLOOP.
      IF SY-SUBRC NE 0.
          TEMSEFIN_TAB-CNTR2 = TMP_CTNR2.
      ENDIF.
      MOVE :
         EXCEL_TAB-EMP_NO    TO TEMSEFIN_TAB-PERNR,
         PAY_BEGDA           TO TEMSEFIN_TAB-FPBEG,
         PAY_ENDDA           TO TEMSEFIN_TAB-FPEND,
         EXCEL_TAB-EMP_NO    TO TEMSEFIN_TAB-PERNR,
         EXCEL_TAB-INCOMETAX TO TEMSEFIN_TAB-INCOMETAX,
         EXCEL_TAB-SURCHARGE TO TEMSEFIN_TAB-SURCHARGE,
         EXCEL_TAB-EDUCESS   TO TEMSEFIN_TAB-EDUCESS,
         EXCEL_TAB-TOTTAX    TO TEMSEFIN_TAB-BETRG,
         EXCEL_TAB-CHKNO     TO TEMSEFIN_TAB-CHKNO,
         EXCEL_TAB-BR_CODE   TO TEMSEFIN_TAB-BRCOD,
         EXCEL_TAB-BANK_CHALLAN  TO TEMSEFIN_TAB-CHLNO,
         N_DATE   TO TEMSEFIN_TAB-CHDAT,
         EXCEL_TAB-PAYPER  TO TEMSEFIN_TAB-FPPER.
* To convert excel_tab-paydate in char10 format to sy-datum format
         CLEAR : N_DATE.
         PERFORM CONV_DATE USING EXCEL_TAB-PAYDATE CHANGING N_DATE.
         MOVE : N_DATE TO TEMSEFIN_TAB-PAYDATE.
         APPEND TEMSEFIN_TAB.
         p_year = EXCEL_TAB-PAYDATE+6(4).
         p_period = EXCEL_TAB-PAYDATE+3(2).
         p_date = EXCEL_TAB-PAYDATE+0(2).
         concatenate p_year p_period p_date INTO PYDTE.
         READ TABLE PAYMENTS WITH KEY TDATE = PYDTE PERNR = EMP_TAB-EMP_NO.
         IF sy-subrc EQ 0.
          MOVE EXCEL_TAB-TOTTAX TO PAYMENTS-TAXPD.
          MODIFY PAYMENTS INDEX sy-tabix.
         ELSE.
          MOVE EXCEL_TAB-TOTTAX TO PAYMENTS-TAXPD.
          APPEND PAYMENTS.
         ENDIF.
         CLEAR: p_year, p_period, p_date, pydte.

  ENDLOOP.
  ENDLOOP.
ENDFORM.                    " get_excel_data

*&---------------------------------------------------------------------*
*&      Form  get_roman_number
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
*      -->P_I  text
*----------------------------------------------------------------------*
FORM GET_ROMAN_NUMBER USING    P_I.
  DATA : COUNTER TYPE I,
         ALPHA(10) TYPE C,
         IQ(10) TYPE C,
         LEN1 TYPE I,
         LOOPCOUNTER TYPE I VALUE 0,
         FINALCOUNTER TYPE I VALUE 0.
  DATA : NUMB(10) TYPE C VALUE '1'.
  DATA : ROMA(150) TYPE C.
  ROMA = TEXT-RMN.
  LEN1 = STRLEN( ROMA ).

  COUNTER = SY-FDPOS + 1.

  NUMB = P_I + 1.
  NUMB = NUMB - 1.

  WHILE LOOPCOUNTER LT NUMB.
    LOOPCOUNTER = LOOPCOUNTER + 1.
    LEN1 = ROMA+0(1).
    LEN1 = LEN1 + 1.
    SHIFT ROMA BY LEN1 PLACES LEFT.
  ENDWHILE.

  LEN1 = ROMA+0(1).

  SHIFT ROMA BY 1 PLACES LEFT.

  WHILE FINALCOUNTER LT LEN1.
    RESULT+FINALCOUNTER = ROMA+FINALCOUNTER(1).
    FINALCOUNTER = FINALCOUNTER + 1.
  ENDWHILE.
ENDFORM.                    " get_roman_number
