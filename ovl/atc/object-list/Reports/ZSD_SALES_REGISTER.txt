  DATA : frmdt(10) TYPE c,

         v_str     TYPE string.

  t_listheader-key = ' '.

    WRITE s_dt-low  TO frmdt USING EDIT MASK '__.__.____'.
    WRITE s_dt-high TO todt  USING EDIT MASK '__.__.____'.
    CONCATENATE 'FROM'
                 frmdt
                 'TO'
                 todt
           INTO v_str SEPARATED BY space.
  ELSEIF s_dt-low NE '00000000' AND  s_dt-high EQ '00000000'      .
    WRITE s_dt-low  TO frmdt USING EDIT MASK '__.__.____'.

                 frmdt
           INTO v_str SEPARATED BY space.
  ELSEIF s_dt-low EQ '00000000' AND  s_dt-high NE '00000000'      .
    WRITE s_dt-high  TO todt USING EDIT MASK '__.__.____'.
    CONCATENATE 'ON'
                 todt
           INTO v_str SEPARATED BY space.
  ENDIF.
  t_listheader-typ = 'S'.
  t_listheader-key = ' '.
  t_listheader-info = v_str.
  APPEND t_listheader.
ENDFORM. " FILL_LISTHEADER
