         todt(10)  TYPE c,
         v_str     TYPE string.


  t_listheader-typ = 'H'.
  t_listheader-key = ' '.
  t_listheader-info = 'PURCHASE REGISTER(POST GST)'.
  APPEND t_listheader.

    WRITE s_budat-low  TO frmdt USING EDIT MASK '__.__.____'.
    WRITE s_budat-high TO todt  USING EDIT MASK '__.__.____'.
    CONCATENATE 'FROM'
                 frmdt
                 'TO'
                 todt
           INTO v_str SEPARATED BY space.
  ELSEIF s_budat-low NE '00000000' AND  s_budat-high EQ '00000000'      .
    WRITE s_budat-low  TO frmdt USING EDIT MASK '__.__.____'.
    CONCATENATE 'ON'
                 frmdt
           INTO v_str SEPARATED BY space.
  ELSEIF s_budat-low EQ '00000000' AND  s_budat-high NE '00000000'      .
    WRITE s_budat-high  TO todt USING EDIT MASK '__.__.____'.
    CONCATENATE 'ON'
                 todt
           INTO v_str SEPARATED BY space.
  ENDIF.
  t_listheader-typ = 'S'.
  t_listheader-key = ' '.
  t_listheader-info = v_str.
  APPEND t_listheader.

*&      Form  FILL_DISPLAYGRID
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
*  -->  p1        text
*  <--  p2        text
*----------------------------------------------------------------------*
FORM fill_displaygrid .
  DATA : lv_srno  TYPE sy-tabix.

  IF it_final1[] IS NOT INITIAL.
    SELECT * FROM t134t
             INTO TABLE lt_t134t
             FOR ALL ENTRIES IN it_final1
             WHERE spras = sy-langu
               AND mtart = it_final1-mtart.

    SELECT lifnr kunnr stcd3
      FROM lfa1
      INTO TABLE it_lfa1
      FOR ALL ENTRIES IN it_final1
      WHERE lifnr = it_final1-lifnr."lfbnr.

****Added for Control code: 14.07.17
    SELECT matnr werks steuc
         FROM marc
         INTO TABLE it_marc
         FOR ALL ENTRIES IN it_final1
         WHERE werks EQ it_final1-werks AND
               matnr EQ it_final1-matnr.
    SELECT *
      FROM ztaxcode
        INTO TABLE it_ztaxcode
        FOR ALL ENTRIES IN it_final1
         WHERE  zmwskz = it_final1-mwskz.

    SELECT * FROM zsd_gstn INTO TABLE it_gstn.

  ENDIF.
  LOOP AT it_final1 INTO wa_final1.
    CLEAR : lwa_t134t.
    READ TABLE lt_t134t INTO lwa_t134t WITH KEY mtart = wa_final1-mtart.
    IF sy-subrc EQ 0.
      wa_final1-mtbez = lwa_t134t-mtbez.
      MODIFY it_final1 FROM wa_final1 TRANSPORTING mtbez.
    ENDIF.
    CLEAR: wa_marc.

    SORT it_marc[] BY matnr.
    READ TABLE it_marc  INTO wa_marc WITH KEY matnr = wa_final1-matnr
                                              werks = wa_final1-werks.
    IF sy-subrc = 0.
      wa_final1-steuc = wa_marc-steuc.
      MODIFY  it_final1 FROM wa_final1 TRANSPORTING steuc.
    ENDIF.

    CLEAR wa_lfa1.
    READ TABLE it_lfa1 INTO wa_lfa1 WITH KEY lifnr = wa_final1-lifnr.
    IF sy-subrc EQ 0.
      wa_final1-stcd3 = wa_lfa1-stcd3.
      wa_final1-regio = wa_final1-stcd3.
      MODIFY  it_final1 FROM wa_final1 TRANSPORTING stcd3 regio.
    ENDIF.
    READ TABLE it_ztaxcode INTO wa_ztaxcode WITH KEY zmwskz = wa_final1-mwskz.
    IF sy-subrc EQ 0.
      wa_final1-rjocgamt = wa_final1-jocgamt.
      wa_final1-rjoigamt = wa_final1-joigamt.
      wa_final1-rjosgamt = wa_final1-josgamt.
      wa_final1-jocgamt = 0.
      wa_final1-joigamt = 0.
      wa_final1-josgamt = 0.
      wa_final1-gross = wa_final1-taxable_val.
      MODIFY  it_final1 FROM wa_final1 TRANSPORTING jocgamt joigamt josgamt rjocgamt rjoigamt rjosgamt gross.
    ENDIF.

    READ TABLE it_gstn INTO wa_gstn WITH KEY werks = wa_final1-werks.
    IF sy-subrc = 0.
      wa_final1-sgstn = wa_gstn-gstn_no.
      MODIFY it_final1 FROM wa_final1 TRANSPORTING sgstn.
    ENDIF.

    lv_srno = sy-tabix.
    wa_final1-srno = lv_srno.
    MODIFY it_final1 FROM wa_final1 TRANSPORTING srno.
    CLEAR : wa_final1.
  ENDLOOP.

  CALL FUNCTION 'REUSE_ALV_GRID_DISPLAY'
    EXPORTING
      i_callback_program     = sy-repid
      i_callback_top_of_page = 'TOP_OF_PAGE'
      is_layout              = fs_layout
      it_fieldcat            = t_fieldcatalog[]
      it_sort                = gt_sort[]
      it_events              = t_event[]
      i_save                 = 'A'
    TABLES
      t_outtab               = it_final1
    EXCEPTIONS
      program_error          = 1
      OTHERS                 = 2.
  IF sy-subrc <> 0.
  ENDIF.
ENDFORM.
