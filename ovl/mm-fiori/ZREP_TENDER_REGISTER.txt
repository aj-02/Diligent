242    **======================================================================*
243    *FORM fetch_tender_data.
244    *
245    *  DATA:
246    *    lt_tender TYPE TABLE OF zmm_pur_tender_d,
247    *    ls_tender TYPE zmm_pur_tender_d,
248    *    ls_tms    TYPE zmm_tms,
249    *    lv_nachn  TYPE pa0002-nachn,
250    *    lv_vorna  TYPE pa0002-vorna,
251    *    lv_ename  TYPE char40.
252    *
253    *  SELECT *
254    *    FROM zmm_pur_tender_d
255    *    INTO TABLE lt_tender
256    *    WHERE submi     IN so_submi
257    *      AND bsart     IN so_bsart
258    *      AND dlg_offic IN so_dlgof
259    *      AND nitdate   IN so_nitdt
260    *      AND tdrsts    IN so_tdrst
261    *      AND mjahr     = p_mjahr.
262    *
263    *  IF sy-subrc <> 0 OR lt_tender IS INITIAL.
264    *    MESSAGE 'No tender records found for the given selection.' TYPE 'I'.
265    *    RETURN.
266    *  ENDIF.
267    *
268    *  LOOP AT lt_tender INTO ls_tender.
269    *
270    *    CLEAR gwa_tdr.
271    *    gwa_tdr-submi     = ls_tender-submi.
272    *    gwa_tdr-bsart     = ls_tender-bsart.
273    *    gwa_tdr-banfn     = ls_tender-banfn.
274    *    gwa_tdr-banfn_pr  = ls_tender-banfn.
275    *    gwa_tdr-workdesc  = ls_tender-workdesc.
276    *    gwa_tdr-tndr_val  = ls_tender-tndr_val.
277    *    gwa_tdr-waers     = ls_tender-waers.
278    *    gwa_tdr-tnndr_typ = ls_tender-tnndr_typ.
279    *    gwa_tdr-dlg_offic = ls_tender-dlg_offic.
280    *    gwa_tdr-tdrtxt    = ls_tender-tdrtxt.
281    *    gwa_tdr-nitdate   = ls_tender-nitdate.
282    *    gwa_tdr-lt_sel_dt = ls_tender-lt_sel_dt.
283    *    gwa_tdr-tdrsts    = ls_tender-tdrsts.
284    *
285    *    SELECT SINGLE *
286    *      FROM zmm_tms
287    *      INTO ls_tms
288    *      WHERE submi = ls_tender-submi.
289    *
290    *    IF sy-subrc = 0.
291    *      IF so_prrdt[] IS NOT INITIAL.
292    *        IF ls_tms-pr_rcpt_dt NOT IN so_prrdt.
293    *          CONTINUE.
294    *        ENDIF.
295    *      ENDIF.
296    *      gwa_tdr-pr_rcpt_dt = ls_tms-pr_rcpt_dt.
297    *      gwa_tdr-noa_dt     = ls_tms-loi_1_act_dt.
298    *      gwa_tdr-emd_recv   = ls_tms-rec_tsal_act_dt.
299    *      gwa_tdr-emd_rtrn   = ls_tms-reo_tsal_stat.
300    *      IF ls_tms-tc_member1 IS NOT INITIAL.
301    *        CLEAR: lv_nachn, lv_vorna, lv_ename.
302    *        SELECT SINGLE vorna nachn
303    *          FROM pa0002
304    *          INTO (lv_vorna, lv_nachn)
305    *          WHERE pernr = ls_tms-tc_member1
306    *            AND endda >= sy-datum
307    *            AND begda <= sy-datum.
308    *        IF sy-subrc = 0.
309    *          CONCATENATE lv_vorna lv_nachn
310    *            INTO lv_ename SEPARATED BY space.
311    *          gwa_tdr-tc_mem_nm = lv_ename.
312    *        ELSE.
313    *          gwa_tdr-tc_mem_nm = ls_tms-tc_member1.
314    *        ENDIF.
315    *      ENDIF.
316    *    ELSE.
317    *      IF so_prrdt[] IS NOT INITIAL.
318    *        CONTINUE.
319    *      ENDIF.
320    *    ENDIF.
321    *
322    *    gv_counter += 1.
323    *    gwa_tdr-slno = gv_counter.
324    *    APPEND gwa_tdr TO gt_tdr.
325    *
326    *  ENDLOOP.
327    *
328    *  IF gt_tdr IS INITIAL.
329    *    MESSAGE 'No records found after applying all filters.' TYPE 'I'.
330    *  ENDIF.

335    **& FORM: FETCH_SO_DATA  (Report 2 - Supplier Order Details Corp & BD)  *
336    **======================================================================*
337    *FORM fetch_so_data.
338    *
339    *  DATA:
340    *    lt_ekko  TYPE TABLE OF ekko,

342    *    lv_banfn LIKE ekpo-banfn,
343    *    lv_txz01 LIKE ekpo-txz01,

345    *    lv_name1 TYPE char40,
346    *    lv_addr  TYPE char80,
347    *    lv_eknam LIKE t024-eknam,
348    *    lv_ename TYPE char40.
349    *
350    *  SELECT *
351    *    FROM ekko
352    *    INTO TABLE lt_ekko
353    *    WHERE ebeln IN so_ebeln
354    *      AND bsart IN so_bsrt2
355    *      AND lifnr IN so_lifnr
356    *      AND bedat IN so_bedat
357    *      AND ekgrp IN so_ekgrp
358    *      AND loekz = ' '.
359    *
360    *  IF sy-subrc <> 0 OR lt_ekko IS INITIAL.
361    *    MESSAGE 'No supply order records found for the given selection.' TYPE 'I'.
362    *    RETURN.
363    *  ENDIF.
364    *
365    *  LOOP AT lt_ekko INTO ls_ekko.
366    *
367    *    CLEAR gwa_so.
368    *
369    *    gwa_so-zzloino    = ls_ekko-zzloino.
370    *    gwa_so-bedat      = ls_ekko-bedat.
371    *    gwa_so-zzsubmi    = ls_ekko-zzsubmi.
372    *    gwa_so-bsart      = ls_ekko-bsart.
373    *    gwa_so-zznat_proc = ls_ekko-zznat_proc.
374    *    gwa_so-zzt_nature = ls_ekko-zzt_nature.
375    *    gwa_so-ebeln      = ls_ekko-ebeln.
376    *    gwa_so-zzvend_cat = ls_ekko-zzvend_cat.
377    *    gwa_so-zzmse      = ls_ekko-zzmse.
378    *    gwa_so-waers      = ls_ekko-waers.
379    *    gwa_so-ktwrt      = ls_ekko-ktwrt.
380    *    gwa_so-wkurs      = ls_ekko-wkurs.
381    *    gwa_so-place_exec = ls_ekko-inco2.
382    *    gwa_so-kdatb      = ls_ekko-kdatb.
383    *    gwa_so-kdate      = ls_ekko-kdate.
384    *
385    *    " Description: prefer EKKO contract description, else first item text
386    *    IF ls_ekko-description IS NOT INITIAL.
387    *      gwa_so-descr = ls_ekko-description.
388    *    ELSE.
389    *      CLEAR lv_txz01.
390    *      SELECT SINGLE txz01
391    *        FROM ekpo INTO lv_txz01
392    *        WHERE ebeln = ls_ekko-ebeln.
393    *      gwa_so-descr = lv_txz01.
394    *    ENDIF.
395    *
396    *    " Contract value in INR
397    *    IF ls_ekko-waers = 'INR'.
398    *      gwa_so-ktwrt_inr = ls_ekko-ktwrt.
399    *    ELSEIF ls_ekko-wkurs > 0.
400    *      gwa_so-ktwrt_inr = ls_ekko-ktwrt * ls_ekko-wkurs.
401    *    ELSE.
402    *      gwa_so-ktwrt_inr = ls_ekko-ktwrt.
403    *    ENDIF.
404    *
405    *    " PR No. and GST from first EKPO item
406    *    CLEAR: lv_banfn, lv_navnw.
407    *    SELECT SINGLE banfn navnw
408    *      FROM ekpo INTO (lv_banfn, lv_navnw)
409    *      WHERE ebeln = ls_ekko-ebeln.
410    *    IF sy-subrc = 0.
411    *      gwa_so-banfn   = lv_banfn.
412    *      gwa_so-gst_amt = lv_navnw.
413    *    ENDIF.
414    *
415    *    " Vendor name and address from LFA1
416    *    PERFORM get_vendor_info
417    *      USING    ls_ekko-lifnr
418    *      CHANGING lv_name1 lv_addr.
419    *    gwa_so-vend_nm   = lv_name1.
420    *    gwa_so-vend_addr = lv_addr.
421    *
422    *    " Indentor name from HR (EKKO-ZZINDPER)
423    *    IF ls_ekko-zzindper IS NOT INITIAL.
424    *      CLEAR lv_ename.
425    *      PERFORM get_emp_name
426    *        USING    ls_ekko-zzindper
427    *        CHANGING lv_ename.
428    *      gwa_so-ind_nm = lv_ename.
429    *    ENDIF.
430    *
431    *    " Dealing officer: purchasing group name from T024
432    *    CLEAR lv_eknam.
433    *    PERFORM get_pgrp_name
434    *      USING    ls_ekko-ekgrp
435    *      CHANGING lv_eknam.
436    *    gwa_so-dlg_nm = lv_eknam.
437    *
438    *    " Amendment data: extend by reading from custom Z amendment table
439    *    " e.g. SELECT ... FROM zcustom_amendment WHERE ebeln = ls_ekko-ebeln.
440    *
441    *    gv_counter += 1.
442    *    gwa_so-slno = gv_counter.
443    *    APPEND gwa_so TO gt_so.
444    *
445    *  ENDLOOP.
446    *
447    *  IF gt_so IS INITIAL.
448    *    MESSAGE 'No supply order records after applying filters.' TYPE 'I'.
449    *  ENDIF.
450    *
451    *ENDFORM.

1162       CLEAR gwa_tdr.

1177       SELECT SINGLE *
1178         FROM zmm_tms
1179         INTO ls_tms

1183         IF so_prrdt[] IS NOT INITIAL.

1190         gwa_tdr-emd_recv   = ls_tms-rec_tsal_act_dt.

1382   *-- RFQ no. = first non-blank EKPO-ANFNR on the document.          "+P4
1383   *-- Contract / OLA no. = first non-blank EKPO-KONNR.               "+P3
1384       CLEAR: lv_txz01, lv_banfn, lv_navnw, lv_first.
1385       LOOP AT lt_item INTO ls_item WHERE ebeln = ls_ekko-ebeln.
1386         IF lv_first = abap_false.
1387           lv_txz01 = ls_item-txz01.
1388           lv_banfn = ls_item-banfn.
1389           lv_navnw = ls_item-navnw.
1390           lv_first = abap_true.
1391         ENDIF.
1392         IF gwa_so-anfnr IS INITIAL AND ls_item-anfnr IS NOT INITIAL.
1393           gwa_so-anfnr = ls_item-anfnr.
1394         ENDIF.
1395         IF gwa_so-konnr IS INITIAL AND ls_item-konnr IS NOT INITIAL.
1396           gwa_so-konnr = ls_item-konnr.
1397         ENDIF.
1398       ENDLOOP.
1399
1400   *-- Effective user: where the document references a contract it is the
1401   *-- CONTRACT's creator, otherwise the document's own creator.       "+P5
1402       CLEAR lv_ernam.
1403       IF gwa_so-konnr IS NOT INITIAL.
1404         READ TABLE lt_cern INTO ls_cern
1405              WITH TABLE KEY ebeln = gwa_so-konnr.
1406         IF sy-subrc = 0.
1407           lv_ernam = ls_cern-ernam.
1408         ENDIF.
1409       ENDIF.
1410       IF lv_ernam IS INITIAL.
1411         lv_ernam = ls_ekko-ernam.
1412       ENDIF.
1413       gwa_so-ernam = lv_ernam.
1414
1415   *-- Apply the user filter here, not in the WHERE clause, because the
1416   *-- value depends on the contract lookup above.  An empty SELECT-OPTION
1417   *-- lets every row through.                                        "+P5
1418       IF lv_ernam NOT IN so_ernam.
1419         CONTINUE.
1420       ENDIF.
1421
1422       " Description: prefer EKKO contract description, else first item text
1423       IF ls_ekko-description IS NOT INITIAL.
1424         gwa_so-descr = ls_ekko-description.
1425       ELSE.
1426         gwa_so-descr = lv_txz01.
1427       ENDIF.
1428
1429       " PR No. and GST from first EKPO item
1430       gwa_so-banfn   = lv_banfn.
1431       gwa_so-gst_amt = lv_navnw.
1432
1433       " Vendor name and address from LFA1
1434       PERFORM get_vendor_info
1435         USING    ls_ekko-lifnr
1436         CHANGING lv_name1 lv_addr.
1437       gwa_so-vend_nm   = lv_name1.
1438       gwa_so-vend_addr = lv_addr.
1439
1440       " Indentor name from HR (EKKO-ZZINDPER)
1441       IF ls_ekko-zzindper IS NOT INITIAL.
1442         CLEAR lv_ename.
1443         PERFORM get_emp_name
1444           USING    ls_ekko-zzindper
1445           CHANGING lv_ename.
1446         gwa_so-ind_nm = lv_ename.
1447       ENDIF.
1448
1449       " Dealing officer: purchasing group name from T024
1450       CLEAR lv_eknam.
1451       PERFORM get_pgrp_name
1452         USING    ls_ekko-ekgrp
1453         CHANGING lv_eknam.
1454       gwa_so-dlg_nm = lv_eknam.
1455
1456       " Amendment data: extend by reading from custom Z amendment table
1457       " e.g. SELECT ... FROM zcustom_amendment WHERE ebeln = ls_ekko-ebeln.
1458
1459   *-- Sl.No. is assigned after the suppression pass below            "+P4
1460       APPEND gwa_so TO gt_so.
1461
1462     ENDLOOP.
1463
1464   *-- Suppress the standalone RFQ row where the referencing PO/contract
1465   *-- is present in this same result set.  An RFQ with no document
1466   *-- against it is kept, so open enquiries are not lost.            "+P4
1467     IF lt_rfq_used IS NOT INITIAL.
1468       LOOP AT gt_so INTO gwa_so WHERE bstyp = 'A'.
1469         lv_tabix = sy-tabix.
1470         READ TABLE lt_rfq_used TRANSPORTING NO FIELDS
1471              WITH TABLE KEY table_line = gwa_so-ebeln.
1472         IF sy-subrc = 0.
1473           DELETE gt_so INDEX lv_tabix.
1474         ENDIF.
1475       ENDLOOP.
1476     ENDIF.
1477
1478   *-- Renumber so Sl.No. stays contiguous after suppression          "+P4
1479     CLEAR gv_counter.
1480     LOOP AT gt_so ASSIGNING <ls_so>.
1481       gv_counter = gv_counter + 1.
1482       <ls_so>-slno = gv_counter.
1483     ENDLOOP.
1484
1485     IF gt_so IS INITIAL.
1486       MESSAGE 'No supply order records after applying filters.' TYPE 'I'.
1487     ENDIF.
1488
1489   ENDFORM.

1498       ls_ekko  TYPE ekko,
1499       lv_name1 TYPE char40,
1500       lv_addr  TYPE char80,
1501       lv_eknam LIKE t024-eknam,

1504

1506       FROM ekko

1508       WHERE ebeln  IN so_ebeln
1509         AND lifnr  IN so_lifnr

1531   *-- cause as Point 2 on the Supply Order report.                  "+P2

1550         USING    ls_ekko-ekgrp
1551         CHANGING lv_eknam.
1552       gwa_gem-dlg_nm = lv_eknam.

1554       " PBG info: check ZMM_TMS if linked via ZZSUBMI

1557           FROM zmm_tms
1558           INTO ls_tms-pbg_rcpt_sch_dt

1560         IF sy-subrc = 0 AND ls_tms-pbg_rcpt_sch_dt IS NOT INITIAL.

1562         ELSE.
