103    *&---------------------------------------------------------------------*
104    *&      Form  F_ENSURE_BATCH_SIGN
105    *&---------------------------------------------------------------------*
106    *  Guarantee every REGUT batch in gt_batch_header has its approver rows
107    *  in ZFI_BATCH_SIGN, keyed to the ACTUAL REGUT key.
108    *
109    *  Why: the DMEE format module ZFI_PAYMEDIUM_DMEE_20 creates the approver
110    *  rows using a PREDICTED LFDNR (SELECT MAX( lfdnr ) + 1) while REGUT is
111    *  not yet committed. When several batches are created in one run the
112    *  prediction repeats, so a later batch's MODIFY overwrites an earlier
113    *  key instead of creating its own - leaving that REGUT batch with no
114    *  approver rows, and it drops out of every worklist. Here we re-create
115    *  any missing rows from ZFI_BNK_RULE against the real REGUT key, so no
116    *  batch can be orphaned. Existing (incl. already-signed) rows are left
117    *  untouched.
118    *----------------------------------------------------------------------*
119    FORM f_ensure_batch_sign USING it_hdr LIKE gt_batch_header.
120      DATA: lt_rule TYPE STANDARD TABLE OF zfi_bnk_rule,
121            ls_rule TYPE zfi_bnk_rule,
122            ls_hdr  TYPE regut,
123            ls_sign TYPE zfi_batch_sign,
124            lt_new  TYPE STANDARD TABLE OF zfi_batch_sign,
125            lv_key  TYPE zfi_batch_sign-batch_no,
126            lv_cnt  TYPE i.
127
128      CHECK it_hdr IS NOT INITIAL.
129
130    * Configured approvers: rule 90700005 = level 1, 90700006 = level 2
131      SELECT * FROM zfi_bnk_rule INTO TABLE lt_rule
132        WHERE zrule = '90700005' OR zrule = '90700006'.
133      IF lt_rule IS INITIAL.
134        RETURN.
135      ENDIF.
136
137      LOOP AT it_hdr INTO ls_hdr.
138        CLEAR lv_key.
139        CONCATENATE ls_hdr-zbukr ls_hdr-banks
140                    ls_hdr-laufd ls_hdr-laufi
141                    ls_hdr-xvorl ls_hdr-dtkey
142                    ls_hdr-lfdnr
143               INTO lv_key RESPECTING BLANKS.
144
145    *   already has approver rows -> leave as is
146        SELECT COUNT(*) INTO lv_cnt FROM zfi_batch_sign WHERE batch_no = lv_key.
147        IF lv_cnt > 0.
148          CONTINUE.
149        ENDIF.
150
151    *   create the approver rows for this batch's paying company code
152        LOOP AT lt_rule INTO ls_rule WHERE zrule_id = ls_hdr-zbukr.
153          CLEAR ls_sign.
154          ls_sign-batch_no = lv_key.
155          ls_sign-signer   = ls_rule-zuser.
156          CASE ls_rule-zrule.
157            WHEN '90700005'. ls_sign-snro = '1'.
158            WHEN '90700006'. ls_sign-snro = '2'.
159            WHEN OTHERS.     CONTINUE.
160          ENDCASE.
161          ls_sign-cdate = sy-datum.
162          ls_sign-ctime = sy-uzeit.
163          APPEND ls_sign TO lt_new.
164        ENDLOOP.
165      ENDLOOP.
166
167      IF lt_new IS NOT INITIAL.
168        INSERT zfi_batch_sign FROM TABLE lt_new ACCEPTING DUPLICATE KEYS.
169        COMMIT WORK.
170      ENDIF.
171    ENDFORM.                    " F_ENSURE_BATCH_SIGN
